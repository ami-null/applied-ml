\begin{frame}{Vectors and matrices}
    \begin{itemize}
        \item Observation $i$ is a \textbf{feature vector} $\mathbf{x}^{(i)} = \big(x_1^{(i)}, x_2^{(i)}, \dots, x_p^{(i)}\big)^\top \in \mathbb{R}^p$ with label $y^{(i)} \in \mathbb{R}$; for example $\mathbf{x}^{(1)} = (230.1,\, 37.8,\, 69.2)^\top$ and $y^{(1)} = 22.1$
        \item Stacking the $n$ observations gives the \textbf{feature matrix} $\mathbf{X} \in \mathbb{R}^{n \times p}$, whose row $i$ is $(\mathbf{x}^{(i)})^\top$, and the \textbf{label vector} $\mathbf{y} \in \mathbb{R}^{n}$
    \end{itemize}
    \[
        \mathbf{X} = \begin{bmatrix} 230.1 & 37.8 & 69.2 \\ 44.5 & 39.3 & 45.1 \\ \vdots & \vdots & \vdots \\ 232.1 & 8.6 & 8.7 \end{bmatrix} = \begin{bmatrix} (\mathbf{x}^{(1)})^\top \\ (\mathbf{x}^{(2)})^\top \\ \vdots \\ (\mathbf{x}^{(n)})^\top \end{bmatrix}, \qquad \mathbf{y} = \begin{bmatrix} 22.1 \\ 10.4 \\ \vdots \\ 13.4 \end{bmatrix}
    \]
    \begin{itemize}
        \item Superscript $(i)$ = which observation (row), subscript $j$ = which feature (column): $x_j^{(i)}$ is entry $(i, j)$ of $\mathbf{X}$; bold lowercase = column vector, bold uppercase = matrix, plain letter = single number
    \end{itemize}
\end{frame}


\begin{frame}{The regression problem}
    \begin{itemize}
        \item Assume the label depends on the features through $y = f(\mathbf{x}) + \varepsilon$: $f$ is a fixed but unknown function and $\varepsilon$ is a random error with mean zero, independent of $\mathbf{x}$
        \item \textbf{Supervised learning}: estimate $f$ from the labelled data $\{(\mathbf{x}^{(i)}, y^{(i)})\}_{i=1}^{n}$, which gives an estimate $\hat{f}$
        \item \textbf{Regression} when the label is quantitative (sales, mpg, balance); classification when it is qualitative (later lectures)
        \item \textbf{Prediction}: compute $\hat{y} = \hat{f}(\mathbf{x})$ for new observations
        \item \textbf{Inference}: understand how the label changes with each feature (which features matter, how large the effects are, whether the relationship is linear)
    \end{itemize}
\end{frame}


\begin{frame}[allowframebreaks]{Advertising data: questions regression can answer}
    \begin{itemize}
        \item Is there a relationship between the advertising budgets and sales, and how strong is it?
        \item Which media are associated with sales, and by how much does each contribute?
        \item How accurately can we predict sales for given budgets?
        \item Is the relationship linear, and do the media reinforce each other (synergy)?
    \end{itemize}

\end{frame}


\section{Simple Linear Regression}


\begin{frame}{Simple linear regression model}
    \begin{itemize}
        \item One feature $x$ and a quantitative label $y$, related by a straight line plus error: $y = \beta_0 + \beta_1 x + \varepsilon$
        \item $\beta_0$ is the \textbf{intercept} (the value of $y$ when $x = 0$), $\beta_1$ is the \textbf{slope} (the change in $y$ for a one-unit increase in $x$); together they are the \textbf{coefficients}
        \item Estimate the coefficients from the training data (hat = estimated): $\hat\beta_0, \hat\beta_1$; predict with $\hat{y} = \hat\beta_0 + \hat\beta_1 x$
        \item \textbf{Fitted value} of observation $i$: $\hat{y}^{(i)} = \hat\beta_0 + \hat\beta_1 x^{(i)}$; \textbf{residual}: $e^{(i)} = y^{(i)} - \hat{y}^{(i)}$; residuals are computed from data, the errors $\varepsilon^{(i)}$ are never observed
        \item Example: $\text{sales} \approx \beta_0 + \beta_1 \times \text{TV}$
    \end{itemize}
\end{frame}


\begin{frame}{Assumptions of the linear regression model}
    \begin{itemize}
        \item For $i = 1, \dots, n$: $y^{(i)} = \beta_0 + \beta_1 x^{(i)} + \varepsilon^{(i)}$, with the features $x^{(i)}$ treated as fixed numbers
        \item \textbf{Linearity, zero-mean errors}: $\mathbb{E}[\varepsilon^{(i)}] = 0$, so $\mathbb{E}[y \mid x] = \beta_0 + \beta_1 x$; this line is the \emph{population regression line}
        \item \textbf{Constant variance, uncorrelated errors}: $\mathrm{Var}(\varepsilon^{(i)}) = \sigma^2$ for every $i$ and $\mathrm{Cov}(\varepsilon^{(i)}, \varepsilon^{(k)}) = 0$ for $i \neq k$
        \item \textbf{Normality}, needed only for exact small-sample tests and intervals: $\varepsilon^{(i)} \sim N(0, \sigma^2)$
        \item Computing $\hat\beta_0, \hat\beta_1$ needs none of these; unbiasedness and the standard errors use the first three; how to check them with residuals is the topic of the next lecture
    \end{itemize}
\end{frame}


\begin{frame}[allowframebreaks]{Least squares criterion}
    \begin{itemize}
        \item \textbf{Residual sum of squares}: $\mathrm{RSS} = \sum_{i=1}^{n} \big(e^{(i)}\big)^2 = \sum_{i=1}^{n} \big(y^{(i)} - \beta_0 - \beta_1 x^{(i)}\big)^2$
        \item \textbf{Least squares} chooses $\hat\beta_0, \hat\beta_1$ to minimize RSS: the line with the smallest total squared vertical distance to the points
        \item Squaring stops positive and negative residuals from cancelling and penalizes large ones more; the red dot marks the minimum (RSS $= 2102.5$ for TV)
    \end{itemize}
    \begin{center}
        \includegraphics[width=0.95\linewidth]{figures/reg_slr_ls.pdf}
    \end{center}
\end{frame}


\begin{frame}{OLS for simple regression: first-order conditions}
    Minimize $\mathrm{RSS}(\beta_0, \beta_1) = \sum_{i=1}^{n} \big(y^{(i)} - \beta_0 - \beta_1 x^{(i)}\big)^2$. At the minimum both partial derivatives are zero:
    \begin{align*}
        \frac{\partial\, \mathrm{RSS}}{\partial \beta_0} &= -2 \sum_{i=1}^{n} \big(y^{(i)} - \beta_0 - \beta_1 x^{(i)}\big) = 0 \\
        \frac{\partial\, \mathrm{RSS}}{\partial \beta_1} &= -2 \sum_{i=1}^{n} x^{(i)} \big(y^{(i)} - \beta_0 - \beta_1 x^{(i)}\big) = 0
    \end{align*}
    Rearranging gives the \textbf{normal equations}, two linear equations in the two unknowns $\beta_0, \beta_1$:
    \begin{align*}
        n\, \beta_0 + \Big(\sum_i x^{(i)}\Big) \beta_1 &= \sum_i y^{(i)} \\
        \Big(\sum_i x^{(i)}\Big) \beta_0 + \Big(\sum_i \big(x^{(i)}\big)^2\Big) \beta_1 &= \sum_i x^{(i)} y^{(i)}
    \end{align*}
\end{frame}


\begin{frame}{OLS for simple regression: solving the normal equations}
    % Write $S_{xx} = \sum_i (x^{(i)} - \bar{x})^2$ and $S_{xy} = \sum_i (x^{(i)} - \bar{x})(y^{(i)} - \bar{y})$.
    \begin{itemize}
        \item Dividing the first normal equation by $n$ gives $\bar{y} = \beta_0 + \beta_1 \bar{x}$, so $\boxed{\hat\beta_0 = \bar{y} - \hat\beta_1 \bar{x}}$
        \item Substituting into the second gives
        \begin{align*}
            \sum_i x^{(i)} (y^{(i)} - \bar{y}) &= \hat\beta_1 \sum_i x^{(i)} (x^{(i)} - \bar{x}) \\
            \hat\beta_1 &= \frac{\sum_i x^{(i)} (y^{(i)} - \bar{y})}{\sum_i x^{(i)} (x^{(i)} - \bar{x})} \\
            &= \frac{\sum_i (x^{(i)} - \bar{x}) (y^{(i)} - \bar{y})}{\sum_i (x^{(i)} - \bar{x})^2} \\
            &= \frac{S_{xy}}{S_{xx}}
        \end{align*}
    \end{itemize}
    % \[
    %     \hat\beta_1 = \frac{S_{xy}}{S_{xx}} = \frac{\sum_{i=1}^{n} \big(x^{(i)} - \bar{x}\big)\big(y^{(i)} - \bar{y}\big)}{\sum_{i=1}^{n} \big(x^{(i)} - \bar{x}\big)^2}, \qquad \hat\beta_0 = \bar{y} - \hat\beta_1 \bar{x}
    % \]
    % \begin{itemize}
    %     \item This is a minimum: the Hessian of RSS is $2 \begin{bmatrix} n & \sum_i x^{(i)} \\ \sum_i x^{(i)} & \sum_i (x^{(i)})^2 \end{bmatrix}$ with determinant $4 n S_{xx} > 0$ unless all $x^{(i)}$ are equal, so RSS is strictly convex and the solution is unique
    % \end{itemize}
\end{frame}


\begin{frame}{Properties of the least squares line}
    \begin{itemize}
        \item The line passes through the point of means $(\bar{x}, \bar{y})$, by the first normal equation; Advertising: $7.033 + 0.04754 \times 147.04 = 14.02 = \bar{y}$
        \item The normal equations say $\sum_i e^{(i)} = 0$ and $\sum_i x^{(i)} e^{(i)} = 0$: the residuals sum to zero and are orthogonal to the feature; consequently $\sum_i \hat{y}^{(i)} e^{(i)} = 0$ and the fitted values have mean $\bar{y}$
        \item The slope is the correlation rescaled: $\hat\beta_1 = r\, s_y / s_x$, where $r$ is the sample correlation and $s_x, s_y$ are the sample standard deviations; Advertising: $0.7822 \times 5.217 / 85.85 = 0.0475$
        \item Since $|r| \le 1$, a one-standard-deviation increase in $x$ goes with an increase of only $r$ standard deviations in $\hat{y}$: the origin of the name \emph{regression} (towards the mean)
    \end{itemize}
\end{frame}


\begin{frame}{Interpreting the coefficients}
    \begin{itemize}
        \item Fitted line for the Advertising data: $\widehat{\text{sales}} = 7.03 + 0.0475 \times \text{TV}$ (sales in thousands of units, TV budget in thousands of dollars)
        \item $\hat\beta_1 = 0.0475$: an additional \$1,000 of TV budget is associated with about $0.0475 \times 1000 = 47.5$ additional units sold
        \item $\hat\beta_0 = 7.03$: about 7,030 units are predicted when the TV budget is zero; the smallest TV budget in the data is \$700, so the intercept sits at the edge of the data
        \item The fit describes association in the observed markets; a causal reading needs more than a fitted line, because budgets were not assigned at random
    \end{itemize}
\end{frame}


\begin{frame}[allowframebreaks]{How accurate are the estimates?}
    \begin{itemize}
        \item The population regression line (red) is the true, unknown relationship; the least squares line (blue) is computed from one sample, and every new sample gives a different line
        \item On average the estimates are correct (unbiased), but a single $\hat\beta_1$ differs from $\beta_1$; the \textbf{standard error} measures the typical size of this difference
    \end{itemize}
    \begin{center}
        \includegraphics[width=0.95\linewidth]{figures/reg_slr_sampling.pdf}
    \end{center}
\end{frame}


\begin{frame}{Mean and variance of the slope estimate}
    \begin{itemize}
        % \item $\hat\beta_1$ is a linear combination of the labels: $\hat\beta_1 = \sum_i w^{(i)} y^{(i)}$ with weights $w^{(i)} = (x^{(i)} - \bar{x})/S_{xx}$, since $\sum_i (x^{(i)} - \bar{x})\bar{y} = 0$
        % \item The weights satisfy $\sum_i w^{(i)} = 0$, $\sum_i w^{(i)} x^{(i)} = 1$ and $\sum_i (w^{(i)})^2 = 1/S_{xx}$
        % \item \textbf{Unbiased}: $\mathbb{E}[\hat\beta_1] = \sum_i w^{(i)} (\beta_0 + \beta_1 x^{(i)}) = \beta_0 \cdot 0 + \beta_1 \cdot 1 = \beta_1$
        % \item \textbf{Variance} (constant variance, uncorrelated errors): $\mathrm{Var}(\hat\beta_1) = \sum_i (w^{(i)})^2 \sigma^2 = \sigma^2 / S_{xx}$
        % \item \textbf{Gauss--Markov}: among all linear unbiased estimators of $\beta_1$, least squares has the smallest variance; with normal errors, $\hat\beta_1 \sim N(\beta_1, \sigma^2 / S_{xx})$ exactly
        \item \textbf{Unbiased}: $\mathbb{E}[\hat\beta_1] = \beta_1$
        \item \textbf{Variance}: $\mathrm{Var}(\hat\beta_1) = \sigma^2 / S_{xx}$
    \end{itemize}
\end{frame}


\begin{frame}{Variance of the estimators}
    \[
        \mathrm{V}(\hat\beta_1) = \frac{\sigma^2}{\sum_{i=1}^{n} \big(x^{(i)} - \bar{x}\big)^2}, \qquad \mathrm{V}(\hat\beta_0) = \sigma^2 \bigg[\frac{1}{n} + \frac{\bar{x}^2}{\sum_{i=1}^{n} \big(x^{(i)} - \bar{x}\big)^2}\bigg]
    \]
    \begin{itemize}
        \item Here $\sigma^2 = \mathrm{Var}(\varepsilon)$ is the variance of the errors; standard errors shrink with more data, less noise, and more spread-out $x^{(i)}$
        \item $\sigma^2$ is unknown and is estimated from the residuals %by $\mathrm{RSE}^2 = \mathrm{RSS}/(n-2)$; replacing $\sigma$ by RSE gives the estimated standard errors reported by software
        \item These formulas assume the errors are uncorrelated with constant variance $\sigma^2$ %(checked with residual plots in the next lecture)
        % \item Advertising, sales on TV: $\mathrm{SE}(\hat\beta_0) = 0.4578$ and $\mathrm{SE}(\hat\beta_1) = 0.0027$
        \item The standard errors (SE) of the estimates are found taking the square roots of the variances
    \end{itemize}
\end{frame}


\begin{frame}{Estimating the error variance}
    \begin{itemize}
        \item The errors $\varepsilon^{(i)}$ are unobserved, so $\sigma^2$ is estimated from the residuals $e^{(i)}$
        % \item The residuals are not free: they satisfy two linear constraints, $\sum_i e^{(i)} = 0$ and $\sum_i x^{(i)} e^{(i)} = 0$, so only $n-2$ of them can vary independently
        \item Recall: \(\mathrm{RSS} = \sum_{i=1}^{n} \big(e^{(i)}\big)^2 = \sum_{i=1}^{n} \big(y^{(i)} - \hat{y}^{(i)}\big)^2\)
        \item It can be shown that $\mathbb{E}[\mathrm{RSS}] = (n-2)\sigma^2$, therefore, $\boxed{\hat\sigma^2 = \mathrm{RSS}/(n-2) = \mathrm{MSE}}$ is an unbiased estimator of \(\sigma^2\) (here, MSE means mean squared error)
        \item The residual standard error (RSE) is then given by \(\mathrm{RSE} = \sqrt{\mathrm{RSS}/(n-2)}\)
        \item Two parameters were estimated from the data, so two degrees of freedom are used up; with $p$ features the divisor becomes $n-p-1$
        \item Advertising, sales on TV: $\mathrm{RSS} = 2102.53$ and $n - 2 = 198$ give $\hat\sigma^2 = 10.62$ and $\mathrm{RSE} = 3.26$
    \end{itemize}
\end{frame}


\begin{frame}{Standard Error of the estimators}
    \[
        \mathrm{SE}(\hat\beta_1) = \sqrt{\frac{\mathrm{MSE}}{\sum_{i=1}^{n} \big(x^{(i)} - \bar{x}\big)^2}}, \qquad \mathrm{SE}(\hat\beta_0) = \sqrt{\mathrm{MSE} \bigg[\frac{1}{n} + \frac{\bar{x}^2}{\sum_{i=1}^{n} \big(x^{(i)} - \bar{x}\big)^2}\bigg]}
    \]
\end{frame}


\begin{frame}{Confidence intervals and hypothesis tests (recap)}
    \begin{itemize}
        \item \textbf{95\% confidence interval} for $\beta_1$: approximately $\hat\beta_1 \pm t_{n-2,\frac{\alpha}{2}} \cdot \mathrm{SE}(\hat\beta_1)$
        \item \textbf{Hypothesis test}: $H_0\colon \beta_1 = 0$ (no relationship) against $H_a\colon \beta_1 \neq 0$, using $t = \hat\beta_1 / \mathrm{SE}(\hat\beta_1)$, which follows a $t_{n-2}$ distribution if $H_0$ is true
        \item The \textbf{$p$-value} is the probability of a $|t|$ at least this large if $H_0$ were true; a small $p$-value (say below 0.05) is evidence against $H_0$
        \item Advertising, sales on TV: 95\% confidence interval for $\beta_1$ is $[0.0422,\, 0.0528]$
    \end{itemize}
    \begin{center}
        \small
        \begin{tabular}{lrrrr}

            \midrule

            TV & 0.0475 & 0.0027 & 17.67 & $<0.0001$ \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}


\begin{frame}{Assessing the fit: RSE and R-squared}
    \begin{itemize}
        \item \textbf{Residual standard error}: $\mathrm{RSE} = \sqrt{\mathrm{RSS}/(n-2)}$, the typical size of a residual, in the units of $y$; Advertising: $\mathrm{RSE} = 3.26$, about 3,260 units (23\% of the mean sales)
        \item \textbf{$R^2$}: the proportion of the variation in $y$ explained by the regression
    \end{itemize}
    \[
        R^2 = 1 - \frac{\mathrm{RSS}}{\mathrm{TSS}}, \qquad \mathrm{TSS} = \sum_{i=1}^{n} \big(y^{(i)} - \bar{y}\big)^2
    \]
    \begin{itemize}
        \item $0 \le R^2 \le 1$; Advertising: $R^2 = 1 - 2102.5/5417.2 = 0.612$, so TV explains about 61\% of the variance in sales
        \item In simple regression $R^2 = r^2$, the squared correlation between $x$ and $y$ ($r = 0.782$)
        \item RSE is in the units of $y$, $R^2$ is unit-free
    \end{itemize}
\end{frame}


\begin{frame}{Analysis of variance: decomposing the total variation}
    \begin{itemize}
        \item Split each deviation from the mean into an explained and an unexplained part: $y^{(i)} - \bar{y} = \big(\hat{y}^{(i)} - \bar{y}\big) + \big(y^{(i)} - \hat{y}^{(i)}\big)$
        \item Square and sum over $i$; the cross term $2 \sum_i (\hat{y}^{(i)} - \bar{y})\, e^{(i)}$ is zero because $\sum_i e^{(i)} = 0$ and $\sum_i \hat{y}^{(i)} e^{(i)} = 0$ (normal equations), which leaves
    \end{itemize}
    \[
        \underbrace{\sum_{i=1}^{n} \big(y^{(i)} - \bar{y}\big)^2}_{\mathrm{TSS}} = \underbrace{\sum_{i=1}^{n} \big(\hat{y}^{(i)} - \bar{y}\big)^2}_{\mathrm{ESS}} + \underbrace{\sum_{i=1}^{n} \big(y^{(i)} - \hat{y}^{(i)}\big)^2}_{\mathrm{RSS}}
    \]
    \begin{itemize}
        \item \textbf{TSS}: total sum of squares (total variation in $y$); \textbf{ESS}: explained (regression) sum of squares; \textbf{RSS}: residual sum of squares
        \item Hence $R^2 = \mathrm{ESS}/\mathrm{TSS} = 1 - \mathrm{RSS}/\mathrm{TSS}$; Advertising, sales on TV: $5417.15 = 3314.62 + 2102.53$
    \end{itemize}
\end{frame}

\section{Multiple Linear Regression}


\begin{frame}{Why not several simple regressions?}
    \begin{center}
        \small
        \begin{tabular}{lrrrr}
            \toprule
            sales on & $\hat\beta_1$ & std. error & $t$-statistic & $p$-value \\
            \midrule
            TV & 0.0475 & 0.0027 & 17.67 & $<0.0001$ \\
            radio & 0.2025 & 0.0204 & 9.92 & $<0.0001$ \\
            newspaper & 0.0547 & 0.0166 & 3.30 & 0.0011 \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{itemize}
        \item Three separate simple regressions give no single prediction from all three budgets, and each one ignores the other media
        \item A simple-regression slope also absorbs the effect of every correlated feature that was left out
        \item Newspaper and radio budgets are correlated ($r = 0.354$): markets that spend more on radio tend to spend more on newspaper, so newspaper may look useful only because it stands in for radio
        \item Remedy: put all features in one model
    \end{itemize}
\end{frame}


\begin{frame}{Multiple linear regression model}
    \begin{itemize}
        \item With $p$ features: $y = \beta_0 + \beta_1 x_1 + \dots + \beta_p x_p + \varepsilon$; $\beta_j$ is the average change in $y$ for a one-unit increase in $x_j$, \emph{holding all other features fixed}
        \item For all $n$ observations at once, $\mathbf{y} = \tilde{\mathbf{X}} \symbf{\beta} + \symbf{\varepsilon}$:
    \end{itemize}
    \[
        \underbrace{\begin{bmatrix} y^{(1)} \\ \vdots \\ y^{(n)} \end{bmatrix}}_{\mathbf{y}}
        = \underbrace{\begin{bmatrix} 1 & x_1^{(1)} & \cdots & x_p^{(1)} \\ \vdots & \vdots & & \vdots \\ 1 & x_1^{(n)} & \cdots & x_p^{(n)} \end{bmatrix}}_{\tilde{\mathbf{X}}}
        \underbrace{\begin{bmatrix} \beta_0 \\ \beta_1 \\ \vdots \\ \beta_p \end{bmatrix}}_{\symbf{\beta}}
        + \underbrace{\begin{bmatrix} \varepsilon^{(1)} \\ \vdots \\ \varepsilon^{(n)} \end{bmatrix}}_{\symbf{\varepsilon}}
    \]
    \begin{itemize}
        \item $\tilde{\mathbf{X}} = [\,\mathbf{1} \;\; \mathbf{X}\,] \in \mathbb{R}^{n \times (p+1)}$ is the feature matrix with a column of ones for the intercept (the \textbf{design matrix}); fitted values $\hat{\mathbf{y}} = \tilde{\mathbf{X}} \hat{\symbf{\beta}}$, residuals $\mathbf{e} = \mathbf{y} - \hat{\mathbf{y}}$
    \end{itemize}
\end{frame}


\begin{frame}{OLS for multiple regression: normal equations}
    Minimize $\mathrm{RSS}(\symbf{\beta}) = \|\mathbf{y} - \tilde{\mathbf{X}} \symbf{\beta}\|^2 = (\mathbf{y} - \tilde{\mathbf{X}} \symbf{\beta})^\top (\mathbf{y} - \tilde{\mathbf{X}} \symbf{\beta})$. Expanding, and using that the scalar $\symbf{\beta}^\top \tilde{\mathbf{X}}^\top \mathbf{y}$ equals its transpose $\mathbf{y}^\top \tilde{\mathbf{X}} \symbf{\beta}$:
    \[
        \mathrm{RSS}(\symbf{\beta}) = \mathbf{y}^\top \mathbf{y} - 2\, \symbf{\beta}^\top \tilde{\mathbf{X}}^\top \mathbf{y} + \symbf{\beta}^\top \tilde{\mathbf{X}}^\top \tilde{\mathbf{X}}\, \symbf{\beta}
    \]
    With the rules $\nabla_{\symbf{\beta}} (\mathbf{a}^\top \symbf{\beta}) = \mathbf{a}$ and $\nabla_{\symbf{\beta}} (\symbf{\beta}^\top A \symbf{\beta}) = 2A\symbf{\beta}$ for symmetric $A$, the gradient is
    \[
        \nabla_{\symbf{\beta}}\, \mathrm{RSS} = -2\, \tilde{\mathbf{X}}^\top \mathbf{y} + 2\, \tilde{\mathbf{X}}^\top \tilde{\mathbf{X}}\, \symbf{\beta} = \mathbf{0} \quad \Longrightarrow \quad \tilde{\mathbf{X}}^\top \tilde{\mathbf{X}}\, \hat{\symbf{\beta}} = \tilde{\mathbf{X}}^\top \mathbf{y}
    \]
    These are the \textbf{normal equations}: $p+1$ linear equations in the $p+1$ unknowns $\beta_0, \dots, \beta_p$.
\end{frame}


\begin{frame}{OLS for multiple regression: solution}
    \begin{itemize}
        \item If $\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}}$ is invertible, the normal equations have the unique solution
    \end{itemize}
    \[
        \hat{\symbf{\beta}} = \big(\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}}\big)^{-1} \tilde{\mathbf{X}}^\top \mathbf{y}
    \]
    % \begin{itemize}
        % \item Invertible $\Leftrightarrow$ the $p+1$ columns of $\tilde{\mathbf{X}}$ are linearly independent: this needs $n \ge p+1$ and no feature that is an exact linear combination of the others
        % \item It is a minimum: the Hessian $2\, \tilde{\mathbf{X}}^\top \tilde{\mathbf{X}}$ is positive definite, since $\mathbf{v}^\top \tilde{\mathbf{X}}^\top \tilde{\mathbf{X}} \mathbf{v} = \|\tilde{\mathbf{X}} \mathbf{v}\|^2 > 0$ for every $\mathbf{v} \neq \mathbf{0}$
        % \item Check with $p = 1$: $\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}} = \begin{bmatrix} n & \sum_i x^{(i)} \\ \sum_i x^{(i)} & \sum_i (x^{(i)})^2 \end{bmatrix}$ and $\tilde{\mathbf{X}}^\top \mathbf{y} = \begin{bmatrix} \sum_i y^{(i)} \\ \sum_i x^{(i)} y^{(i)} \end{bmatrix}$ are exactly the simple regression normal equations
        % \item Software solves the normal equations with a QR or SVD decomposition rather than an explicit inverse, which is numerically more stable
    % \end{itemize}
\end{frame}

\begin{frame}{Sampling distribution of the OLS estimator}
    \begin{itemize}
        \item Substitute $\mathbf{y} = \tilde{\mathbf{X}} \symbf{\beta} + \symbf{\varepsilon}$ into the solution: $\hat{\symbf{\beta}} = \symbf{\beta} + (\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1} \tilde{\mathbf{X}}^\top \symbf{\varepsilon}$, so $\hat{\symbf{\beta}}$ is the truth plus a linear function of the errors
        \item \textbf{Unbiased}: $\mathbb{E}[\symbf{\varepsilon}] = \mathbf{0}$ gives $\mathbb{E}[\hat{\symbf{\beta}}] = \symbf{\beta}$
        \item \textbf{Covariance}: with $\mathrm{Cov}(\symbf{\varepsilon}) = \sigma^2 \mathbf{I}$,
    \end{itemize}
    \[
        \mathrm{Cov}(\hat{\symbf{\beta}}) = (\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1} \tilde{\mathbf{X}}^\top (\sigma^2 \mathbf{I}) \tilde{\mathbf{X}} (\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1} = \sigma^2 (\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1}
    \]
    \begin{itemize}
        \item If errors follow the normal distribution, then $\hat{\symbf{\beta}} \sim N\big(\symbf{\beta}, \sigma^2 (\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1}\big)$; $\mathrm{SE}(\hat\beta_j) = \hat\sigma \sqrt{[(\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1}]_{jj}}$ with $\hat\sigma^2 = \mathrm{RSS}/(n-p-1)$, and $t_j = \hat\beta_j / \mathrm{SE}(\hat\beta_j) \sim t_{n-p-1}$ under $H_0\colon \beta_j = 0$
    \end{itemize}
\end{frame}


\mode<article>{
    \paragraph{Why the divisor is $n-p-1$.}
    The residual vector is $\mathbf{e} = (\mathbf{I} - \mathbf{H})\mathbf{y} = (\mathbf{I} - \mathbf{H})\symbf{\varepsilon}$, because $(\mathbf{I} - \mathbf{H})\tilde{\mathbf{X}} = \mathbf{0}$. Since $\mathbf{I} - \mathbf{H}$ is symmetric and idempotent, $\mathrm{RSS} = \mathbf{e}^\top \mathbf{e} = \symbf{\varepsilon}^\top (\mathbf{I} - \mathbf{H}) \symbf{\varepsilon}$, and therefore $\mathbb{E}[\mathrm{RSS}] = \sigma^2\, \mathrm{tr}(\mathbf{I} - \mathbf{H}) = \sigma^2 (n - p - 1)$. Dividing RSS by $n-p-1$ thus gives an unbiased estimator of $\sigma^2$.
}

\begin{frame}{Is at least one feature useful? The F-test}
    \begin{itemize}
        \item $H_0\colon \beta_1 = \beta_2 = \dots = \beta_p = 0$ against $H_a\colon$ at least one $\beta_j \neq 0$
    \end{itemize}
    \[
        F = \frac{(\mathrm{TSS} - \mathrm{RSS})/p}{\mathrm{RSS}/(n-p-1)} = \frac{\mathrm{ESS}/p}{\mathrm{RSS}/(n-p-1)}
    \]
    \begin{itemize}
        \item If $H_0$ is true, $F$ is close to 1; a value much larger than 1 is evidence against $H_0$, with the $p$-value taken from the $F_{p,\,n-p-1}$ distribution
        \item Advertising: $F = 570.3$ on $(3, 196)$ degrees of freedom, $p < 0.0001$: at least one medium is related to sales
        \item Why not just look at the individual $t$-tests? With many features, about 5\% of the $p$-values fall below 0.05 by chance even if no feature matters; the $F$-test accounts for the number of features
        \item Deciding \emph{which} features to keep is variable selection, deferred to a later lecture
    \end{itemize}
\end{frame}

\section{Prediction Interval}


\begin{frame}{Prediction and its uncertainty}
    \begin{itemize}
        \item For a new observation $\mathbf{x}^\ast = (x_1^\ast, \dots, x_p^\ast)^\top$ that is not in the training data, with $\tilde{\mathbf{x}}^\ast = (1, x_1^\ast, \dots, x_p^\ast)^\top$, the prediction is
    \end{itemize}
    \[
        \hat{y}^\ast = \hat\beta_0 + \hat\beta_1 x_1^\ast + \dots + \hat\beta_p x_p^\ast = (\tilde{\mathbf{x}}^\ast)^\top \hat{\symbf{\beta}}
    \]
    \begin{itemize}
        \item Uncertainty from estimating the coefficients (reducible): $\hat{f}$ differs from the true $f$; the \textbf{confidence interval} covers the \emph{average} response $f(\mathbf{x}^\ast)$ over many observations with these features
        \item Model bias: a linear model is an approximation of reality; the intervals assume it is correct (checked with residual plots in the next lecture)
        \item Irreducible error: a single response is $y^\ast = f(\mathbf{x}^\ast) + \varepsilon^\ast$ even if $f$ were known; the \textbf{prediction interval} covers an \emph{individual} response and is always wider than the confidence interval
    \end{itemize}
\end{frame}


\begin{frame}{Confidence and prediction intervals: formulas}
    \begin{itemize}
        \item The variance of the prediction $\hat{y}^\ast = (\tilde{\mathbf{x}}^\ast)^\top \hat{\symbf{\beta}}$ follows from $\mathrm{Cov}(\hat{\symbf{\beta}}) = \sigma^2 (\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1}$:
    \end{itemize}
    \[
        \mathrm{Var}(\hat{y}^\ast) = \sigma^2\, (\tilde{\mathbf{x}}^\ast)^\top (\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1} \tilde{\mathbf{x}}^\ast
    \]
    \begin{itemize}
        \item 95\% \textbf{confidence interval} for $f(\mathbf{x}^\ast)$: $\hat{y}^\ast \pm t_{0.975,\, n-p-1}\, \hat\sigma \sqrt{(\tilde{\mathbf{x}}^\ast)^\top (\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1} \tilde{\mathbf{x}}^\ast}$
        \item 95\% \textbf{prediction interval} for $y^\ast$: $\hat{y}^\ast \pm t_{0.975,\, n-p-1}\, \hat\sigma \sqrt{1 + (\tilde{\mathbf{x}}^\ast)^\top (\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1} \tilde{\mathbf{x}}^\ast}$; the extra 1 under the root is the variance of the new error $\varepsilon^\ast$
        \item The prediction interval never shrinks below $\pm t_{0.975}\, \hat\sigma$, however much data we have; the confidence interval shrinks to zero width as $n$ grows, and both are widest for $\mathbf{x}^\ast$ far from the centre of the data
    \end{itemize}
\end{frame}

    \end{itemize}
\end{frame}

\section{Interaction}


\begin{frame}{Interactions: relaxing the additive assumption}
    \begin{itemize}
        \item The additive model assumes the effect of TV on sales is the same at every radio budget; spending on both together may work better than the sum of the parts (synergy)
        \item Add an \textbf{interaction term}, a new column of $\tilde{\mathbf{X}}$ with entries $x_1^{(i)} x_2^{(i)}$: $y = \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \beta_3 x_1 x_2 + \varepsilon = \beta_0 + \beta_1 x_1 + (\beta_2 + \beta_3 x_1)\, x_2 + \varepsilon$, so the effect of $x_2$ depends on $x_1$
    \end{itemize}
\end{frame}


\begin{frame}{Interaction: example}
    \begin{itemize}
        \item Advertising: $\widehat{\text{sales}} = 6.75 + 0.0191\,\text{TV} + 0.0289\,\text{radio} + 0.0011\,\text{TV} \times \text{radio}$; the interaction has $p < 0.0001$, $R^2$ rises from 0.897 to 0.968 and RSE falls from 1.68 to 0.94
        \item An extra \$1,000 on radio is associated with about 28.9 more units when TV $= 0$ and about 137.5 more units when TV $= \$100{,}000$
        \item \textbf{Hierarchical principle}: if an interaction is in the model, keep the main effects even when their $p$-values are large
    \end{itemize}
\end{frame}
