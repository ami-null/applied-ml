\section{Introduction to Machine Learning}

\begin{frame}{Introduction}
	\begin{itemize}
		\item Your phone recognizes your face to unlock itself.
		\item Netflix and YouTube suggest what to watch next based on your viewing history.
		\item Gmail quietly moves suspicious messages into your spam folder.
		\item Google Maps predicts how long your commute will take, using real-time and historical traffic data.
		\item Your bank flags an unusual transaction on your card as possible fraud.
	\end{itemize}

	\vspace{0.5em}
	All of these are machine learning systems working in the background --- none of them were explicitly programmed with a rule for every situation; they learned from data.
\end{frame}


\begin{frame}{What is Machine Learning?}
	Tom Mitchell's formal definition: a program learns from experience $E$ with respect to task $T$ and performance measure $P$, if its performance at $T$, measured by $P$, improves with $E$.

	\vspace{0.4em}
	\textbf{Example (spam filter)}:
	\begin{itemize}
		\item $E$ (experience): a collection of past emails, each already labeled spam or not spam.
		\item $T$ (task): classifying a new, incoming email as spam or not spam.
		\item $P$ (performance measure): the percentage of emails correctly classified.
		\item As the filter sees more labeled emails ($E$), its classification accuracy ($P$) on new emails improves --- that improvement \emph{is} the learning.
	\end{itemize}
\end{frame}

\begin{frame}{Inputs, Outputs, and Terminology}
    \begin{itemize}
        \item \textbf{Motivating Example}: Predict product \emph{sales} ($Y$) based on advertising budgets spent across three media: \emph{TV} ($X_1$), \emph{radio} ($X_2$), and \emph{newspaper} ($X_3$).
        \vspace{0.4em}
        \item \textbf{Input Variables ($X$)}:
        \begin{itemize}
            \item Represented as a vector $X = (X_1, X_2, \dots, X_p)$.
            \item Synonyms: \textbf{predictors}, \textbf{independent variables}, \textbf{features}, or \textbf{covariates}.
        \end{itemize}
        \vspace{0.4em}
        \item \textbf{Output Variable ($Y$)}:
        \begin{itemize}
            \item The target quantity we wish to estimate or predict.
            \item Synonyms: \textbf{response} or \textbf{dependent variable} or \textbf{target}.
        \end{itemize}
    \end{itemize}
\end{frame}


\begin{frame}{Formulating the Problem: $Y = f(X) + \epsilon$}
    We assume a general relationship between the response $Y$ and predictors $X$:
    $$Y = f(X) + \epsilon$$

    \begin{itemize}
        \item \textbf{Systematic Function ($f$)}:
        \begin{itemize}
            \item A fixed but \emph{unknown} function connecting $X_1, \dots, X_p$ to $Y$.
            \item Represents the systematic information that inputs provide about the output.
        \end{itemize}
        \vspace{0.4em}
        \item \textbf{Random Error Term ($\epsilon$)}:
        \begin{itemize}
            \item Captures unmeasured factors, noise, or inherent variation.
            \item Assumed to be independent of $X$ with a mean of zero ($E[\epsilon] = 0$).
        \end{itemize}
    \end{itemize}
\end{frame}


\begin{frame}{The Core Goal: Estimating $f$}
    \begin{itemize}
        \item Because the true function $f$ is unknown, we must use observed data to compute an estimate $\hat{f}$.
        \item \textbf{Statistical / Machine Learning}: A collection of data-driven approaches for estimating $f$ to obtain predictions:
        $$\hat{Y} = \hat{f}(X)$$
        \item \textbf{Two Main Goals of Estimating $f$}:
        \begin{enumerate}
            \item \textbf{Prediction}: Make accurate predictions of $\hat{Y}$ for new, unseen observations of $X$.
            \item \textbf{Inference}: Understand the relationship between $X$ and $Y$ (e.g., \emph{Which media budgets actually drive sales?}).
        \end{enumerate}
    \end{itemize}
\end{frame}


\section{Relationship Between Machine Learning, Statistics, and Data Science}


\begin{frame}{Statistics}
	\begin{itemize}
		\item \textbf{Statistics}: the mathematical discipline of collecting, analyzing, and interpreting data, with heavy emphasis on \emph{inference} --- quantifying uncertainty, testing hypotheses, estimating confidence intervals.
		\item Emphasis on mathematical inference, hypothesis testing, and quantifying uncertainty.
		\item A statistician cares whether a coefficient is significant and interpretable; an ML engineer mostly cares whether the model predicts well on new data, even if it is a black box.
	\end{itemize}
\end{frame}


\begin{frame}{Machine Learning}
	\begin{itemize}
		\item \textbf{Machine Learning}: algorithms that learn patterns from data primarily to make \emph{predictions} on new, unseen data.
		\item Emphasis on predictive algorithms that generalise well to unseen data.
		\item Statistics supplies much of ML's theoretical foundation (linear regression, maximum likelihood estimation, hypothesis testing).
		\item ML is one tool a data scientist uses, alongside dashboards, ETL pipelines, and A/B testing.
	\end{itemize}
\end{frame}


\begin{frame}{Data Science}
	\begin{itemize}
		\item \textbf{Data Science}: a broad, applied field that combines domain knowledge, programming/engineering, statistics, and ML to go from raw data to business decisions --- includes data collection, cleaning, visualization, and communicating results to stakeholders.
		\item An interdisciplinary umbrella combining domain knowledge, software engineering, stats, and ML to solve end-to-end problems.
		\item A data scientist at a company typically needs a bit of both statistics and ML, as well as the ability to explain results to a non-technical manager.
	\end{itemize}
\end{frame}

\section{Types of Machine Learning}


\begin{frame}{Supervised Learning}
	Learning a mapping from input \textbf{features} $X$ (the measurable inputs, e.g., square footage, pixel values) to an output \textbf{label} $Y$ (the ``correct answer''), using a \textbf{training set} of examples where $Y$ is already known.

	\begin{itemize}
		\item \textbf{Classification}: label is a discrete category. Example: is this email spam or not? Does this X-ray show a tumor or not (used in hospital diagnostic-support tools)?
		\item \textbf{Regression}: label is a continuous number. Example: predicting a house's sale price from its size, location, and age; predicting tomorrow's temperature.
		\item The \textbf{model} is the learned function $\hat{Y} = f(X)$; a \textbf{loss function} measures how wrong its predictions are, and training adjusts the model to minimize this loss on the training data while still \textbf{generalizing} well to new, unseen data (avoiding \textbf{overfitting}, i.e., memorizing training data instead of learning general patterns).
	\end{itemize}
\end{frame}


\begin{frame}{Supervised Learning: Examples}
	\begin{itemize}
		\item Everyday examples: your email's spam filter, your phone unlocking with face recognition, autocorrect/predictive text.
		\item Industry examples: banks predicting loan default risk (credit scoring), Tesla/Waymo detecting pedestrians and stop signs from camera images, hospitals using ML to flag likely-cancerous scans for radiologist review.
	\end{itemize}
\end{frame}


\begin{frame}{Unsupervised Learning}
	Finding structure or patterns in data that has \textbf{no labels} --- there is no ``correct answer'' provided, the algorithm must discover structure on its own.

	\begin{itemize}
		\item \textbf{Clustering}: grouping similar data points together (e.g., $k$-means). Example: a retailer segments customers into groups by purchasing behavior, without being told in advance what the groups should be, in order to target each group with different marketing.
		\item \textbf{Dimensionality reduction}: compressing many features into a few that capture most of the information (e.g., PCA), useful for visualization or as a preprocessing step.
		\item \textbf{Anomaly detection}: flagging data points that don't fit the usual pattern. Example: banks flagging an unusual credit card transaction as possible fraud.
	\end{itemize}
\end{frame}


\begin{frame}{Unsupervised Learning: Examples}
	\begin{itemize}
		\item Google Photos automatically grouping photos of the same person's face together, even though it was never told who that person is.
		\item Genomics researchers clustering gene-expression data to discover previously unknown subtypes of a disease.
	\end{itemize}
\end{frame}


\begin{frame}{Reinforcement Learning}
	An \textbf{agent} learns to make a sequence of decisions by interacting with an \textbf{environment}: at each \textbf{state}, it takes an \textbf{action}, receives a \textbf{reward} (or penalty), and updates its \textbf{policy} (its strategy for choosing actions) to maximize cumulative reward over time.

	\vspace{0.3em}

	\begin{center}
		\begin{tikzpicture}[node distance=2.2cm, font=\small]
			\node[draw, rounded corners, fill=blue!10, minimum width=2.2cm, minimum height=1cm] (agent) {Agent};

			\draw[->] (env.south) -- ++(0,-0.6) -| node[below, pos=0.25]{state, reward} (agent.south);
		\end{tikzpicture}
	\end{center}
\end{frame}


\begin{frame}{Reinforcement Learning (cont.)}
	\begin{itemize}
		\item Unlike supervised learning, there is no fixed ``correct'' action for each state --- the agent must learn from trial and error, and rewards can be delayed (an action now may only pay off many steps later).
		\item Analogy: training a dog with treats --- you don't tell it the exact muscle movements for ``sit,'' you reward it when it happens to get closer to the desired behavior.
		\item Example: DeepMind's AlphaGo learned to play Go well enough to beat the world champion, largely through self-play reinforcement learning.
	\end{itemize}
\end{frame}


\begin{frame}{Reinforcement Learning: Examples}
	\begin{itemize}
		\item Robots learning to walk or grasp objects through repeated physical trial and error.
		\item Google used reinforcement learning to reduce the energy used for cooling its data centers.
		\item YouTube uses RL-based methods to optimize recommendations for long-term viewer engagement rather than just the next click.
	\end{itemize}
\end{frame}


\begin{frame}{Semi-Supervised Learning}
    \textbf{Semi-Supervised Learning}: Bridges the gap between supervised and unsupervised learning by leveraging a \emph{small set} of labeled data alongside a \emph{large pool} of unlabeled data.
	\begin{itemize}
		\item Drastically reduces human labeling cost while boosting model accuracy.
		\item \emph{Example}: Medical diagnostics where only a tiny fraction of MRI scans are labeled by expert radiologists.
	\end{itemize}
\end{frame}


\begin{frame}{Self-Supervised Learning}
	\textbf{Self-Supervised Learning}: The model generates its own labels directly from the input data structure without human annotation.
	\begin{itemize}
		\item Hides part of the data and trains the model to predict the missing pieces (pretext task).
		\item Powers modern Large Language Models (LLMs) and foundation vision models.
		\item \emph{Example}: Training ChatGPT by predicting the next word in billions of sentences, or predicting masked patches in images.
	\end{itemize}
\end{frame}


\section{Applications of Machine Learning}


\begin{frame}{Applications in Everyday Life}
	\begin{itemize}
		\item \textbf{Voice assistants}: Siri, Alexa, and Google Assistant use ML for speech recognition and natural language understanding.
		\item \textbf{Recommendations}: Netflix suggesting what to watch, Spotify's Discover Weekly, YouTube's homepage, Amazon's ``customers also bought.''
		\item \textbf{Navigation}: Google Maps predicting traffic and estimated arrival time from historical and real-time location data.
		\item \textbf{Photography and communication}: face unlock on smartphones, autocorrect and predictive text, spam filtering in email, automatic photo tagging.
		\item \textbf{Social media}: automatically filtering out content, ranking your feed, and suggesting people you may know.
	\end{itemize}
\end{frame}


\begin{frame}[allowframebreaks]{Applications in Industry and Organizations}
	\begin{itemize}
	\item \textbf{Healthcare}: diagnostic support from medical images (X-rays, MRIs); DeepMind's AlphaFold predicting 3D protein structures, accelerating drug discovery.
	\item \textbf{Finance}: fraud detection on card transactions, algorithmic trading, automated credit scoring for loan approval.
	\item \textbf{Manufacturing}: predictive maintenance --- companies like Siemens and GE use sensor data to predict when a machine will fail \emph{before} it breaks down, avoiding costly downtime.
	\item \textbf{Retail}: demand forecasting and inventory management (e.g., Amazon predicting how much of a product to stock in each warehouse).
	\item \textbf{Transportation}: self-driving vehicles (Tesla, Waymo), and dynamic/surge pricing for ride-sharing (Uber, Lyft) based on real-time demand.
	\item \textbf{Agriculture}: predicting crop yields and detecting plant disease from drone/satellite imagery for precision farming.
	\item \textbf{Environmental Science}: weather forecasting and climate change modeling.
	\end{itemize}
\end{frame}
