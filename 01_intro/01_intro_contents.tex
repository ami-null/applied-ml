\section{Introduction to Machine Learning}

\begin{frame}{What is Machine Learning?}
	Tom Mitchell's formal definition: a program learns from experience $E$ with respect to task $T$ and performance measure $P$, if its performance at $T$, measured by $P$, improves with $E$.

	\begin{itemize}
		\item Instead of a human writing explicit rules, we give the algorithm examples (data), and it discovers the rules (a \textbf{model}) on its own.
		\item Example: a spam filter written the traditional way needs rules like ``block if subject contains `free money'.'' An ML spam filter is instead shown thousands of emails labeled spam / not spam, and it learns the patterns itself --- including patterns a human would never think to hard-code.
		\item This is the key shift: traditional programming is \emph{rules + data $\to$ output}; machine learning is \emph{data + output $\to$ rules}.
	\end{itemize}

\begin{frame}{Scope: AI, Machine Learning, and Deep Learning}
	\begin{itemize}
		\item \textbf{Artificial Intelligence (AI)}: any technique that lets a machine mimic intelligent behavior --- this includes hand-coded expert systems (e.g., a chess engine with programmed rules) and does not require learning from data at all.
		\item \textbf{Machine Learning (ML)}: the subset of AI where the system \emph{learns} its behavior from data rather than being explicitly programmed.
		\item \textbf{Deep Learning (DL)}: the subset of ML that uses multi-layered neural networks, especially effective for unstructured data like images, audio, and text (e.g., ChatGPT, face recognition on your phone).
	\end{itemize}

\begin{frame}{Machine Learning, Statistics, and Data Science}
	\begin{itemize}
		\item \textbf{Statistics}: the mathematical discipline of collecting, analyzing, and interpreting data, with heavy emphasis on \emph{inference} --- quantifying uncertainty, testing hypotheses, estimating confidence intervals.
		\item \textbf{Data Science}: a broad, applied field that combines domain knowledge, programming/engineering, statistics, and ML to go from raw data to business decisions --- includes data collection, cleaning, visualization, and communicating results to stakeholders.
		\item \textbf{Machine Learning}: algorithms that learn patterns from data primarily to make \emph{predictions} on new, unseen data.
		\item Statistics supplies much of ML's theoretical foundation (linear regression, maximum likelihood estimation, hypothesis testing); ML is one tool a data scientist uses, alongside dashboards, ETL pipelines, and A/B testing.
		\item A statistician cares whether a coefficient is significant and interpretable; an ML engineer mostly cares whether the model predicts well on new data, even if it is a black box; a data scientist at a company typically needs a bit of both, plus the ability to explain results to a non-technical manager.
	\end{itemize}

\section{Types of Machine Learning}

\begin{frame}{Supervised Learning}
	Learning a mapping from input \textbf{features} $X$ (the measurable inputs, e.g., square footage, pixel values) to an output \textbf{label} $Y$ (the ``correct answer''), using a \textbf{training set} of examples where $Y$ is already known.

	\begin{itemize}
		\item \textbf{Classification}: label is a discrete category. Example: is this email spam or not? Does this X-ray show a tumor or not (used in hospital diagnostic-support tools)?
		\item \textbf{Regression}: label is a continuous number. Example: predicting a house's sale price from its size, location, and age; predicting tomorrow's temperature.
		\item The \textbf{model} is the learned function $\hat{Y} = f(X)$; a \textbf{loss function} measures how wrong its predictions are, and training adjusts the model to minimize this loss on the training data while still \textbf{generalizing} well to new, unseen data (avoiding \textbf{overfitting}, i.e., memorizing training data instead of learning general patterns).
		\item Everyday examples: your email client's spam filter (Gmail), your phone unlocking with face recognition, autocorrect/predictive text.
		\item Industry examples: banks predicting loan default risk (credit scoring), Tesla/Waymo detecting pedestrians and stop signs from camera images, hospitals using ML to flag likely-cancerous scans for radiologist review.
	\end{itemize}
\end{frame}

\begin{frame}{Unsupervised Learning}
	Finding structure or patterns in data that has \textbf{no labels} --- there is no ``correct answer'' provided, the algorithm must discover structure on its own.

	\begin{itemize}
		\item \textbf{Clustering}: grouping similar data points together (e.g., $k$-means). Example: a retailer segments customers into groups by purchasing behavior, without being told in advance what the groups should be, in order to target each group with different marketing.
		\item \textbf{Dimensionality reduction}: compressing many features into a few that capture most of the information (e.g., PCA), useful for visualization or as a preprocessing step.
		\item \textbf{Anomaly detection}: flagging data points that don't fit the usual pattern. Example: banks flagging an unusual credit card transaction as possible fraud.
		\item Everyday example: Google Photos automatically grouping photos of the same person's face together, even though it was never told who that person is.
		\item Industry example: genomics researchers clustering gene-expression data to discover previously unknown subtypes of a disease.
	\end{itemize}
\end{frame}

\begin{frame}{Reinforcement Learning (Brief Overview)}
	An \textbf{agent} learns to make a sequence of decisions by interacting with an \textbf{environment}: at each \textbf{state}, it takes an \textbf{action}, receives a \textbf{reward} (or penalty), and updates its \textbf{policy} (its strategy for choosing actions) to maximize cumulative reward over time.

	\vspace{0.3em}

	\begin{center}
		\begin{tikzpicture}[node distance=2.2cm, font=\small]
			\node[draw, rounded corners, fill=blue!10, minimum width=2.2cm, minimum height=1cm] (agent) {Agent};
			\node[draw, rounded corners, fill=green!10, minimum width=2.6cm, minimum height=1cm, right=3.2cm of agent] (env) {Environment};
			\draw[->] (agent.north) -- ++(0,0.6) -| node[above, pos=0.25]{action} (env.north);
			\draw[->] (env.south) -- ++(0,-0.6) -| node[below, pos=0.25]{state, reward} (agent.south);
		\end{tikzpicture}
	\end{center}

	\begin{itemize}
		\item Unlike supervised learning, there is no fixed ``correct'' action for each state --- the agent must learn from trial and error, and rewards can be delayed (an action now may only pay off many steps later).
		\item Analogy: training a dog with treats --- you don't tell it the exact muscle movements for ``sit,'' you reward it when it happens to get closer to the desired behavior.
		\item Example: DeepMind's AlphaGo learned to play Go well enough to beat the world champion, largely through self-play reinforcement learning.
		\item Example: robots learning to walk or grasp objects through repeated physical trial and error.
		\item Industry example: Google used reinforcement learning to reduce the energy used for cooling its data centers; YouTube uses RL-based methods to optimize recommendations for long-term viewer engagement rather than just the next click.
	\end{itemize}
\end{frame}

\section{Applications of Machine Learning}

\begin{frame}{Applications in Everyday Life}
	\begin{itemize}
		\item \textbf{Voice assistants}: Siri, Alexa, and Google Assistant use ML for speech recognition and natural language understanding.
		\item \textbf{Recommendations}: Netflix suggesting what to watch, Spotify's Discover Weekly, YouTube's homepage, Amazon's ``customers also bought.''
		\item \textbf{Navigation}: Google Maps predicting traffic and estimated arrival time from historical and real-time location data.
		\item \textbf{Photography and communication}: face unlock on smartphones, autocorrect and predictive text, spam filtering in email, automatic photo tagging.
		\item \textbf{Social media}: automatically filtering out content, ranking your feed, and suggesting people you may know.
	\end{itemize}
\end{frame}

\begin{frame}{Applications in Industry and Organizations}
	\begin{itemize}
	\item \textbf{Healthcare}: diagnostic support from medical images (X-rays, MRIs); DeepMind's AlphaFold predicting 3D protein structures, accelerating drug discovery.
	\item \textbf{Finance}: fraud detection on card transactions, algorithmic trading, automated credit scoring for loan approval.
	\item \textbf{Manufacturing}: predictive maintenance --- companies like Siemens and GE use sensor data to predict when a machine will fail \emph{before} it breaks down, avoiding costly downtime.
	\item \textbf{Retail}: demand forecasting and inventory management (e.g., Amazon predicting how much of a product to stock in each warehouse).
	\item \textbf{Transportation}: self-driving vehicles (Tesla, Waymo), and dynamic/surge pricing for ride-sharing (Uber, Lyft) based on real-time demand.
	\item \textbf{Agriculture}: predicting crop yields and detecting plant disease from drone/satellite imagery for precision farming.
	\end{itemize}
\end{frame}
