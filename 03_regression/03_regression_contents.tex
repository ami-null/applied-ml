\section{Data as Vectors and Matrices}

\begin{frame}{Observations, features and labels}
    \begin{center}
        \small
        \begin{tabular}{crrrr}
            \toprule
            & \multicolumn{3}{c}{features} & label \\
            \cmidrule(lr){2-4} \cmidrule(lr){5-5}
            $i$ & TV & radio & newspaper & sales \\
            \midrule
            1 & 230.1 & 37.8 & 69.2 & 22.1 \\
            2 & 44.5 & 39.3 & 45.1 & 10.4 \\
            $\vdots$ & $\vdots$ & $\vdots$ & $\vdots$ & $\vdots$ \\
            200 & 232.1 & 8.6 & 8.7 & 13.4 \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{itemize}
        \item Each row is one \textbf{observation} (a market): $n = 200$ observations
        \item The columns TV, radio, newspaper are the \textbf{features} (predictors, inputs, independent variables): $p = 3$, budgets in thousands of dollars
        \item The column sales is the \textbf{label} (response, output, target, dependent variable): what we want to predict, in thousands of units
    \end{itemize}
\end{frame}

\begin{frame}{Vectors and matrices}
    \begin{itemize}
        \item Observation $i$ is a \textbf{feature vector} $\mathbf{x}^{(i)} = \big(x_1^{(i)}, x_2^{(i)}, \dots, x_p^{(i)}\big)^\top \in \mathbb{R}^p$ with label $y^{(i)} \in \mathbb{R}$; for example $\mathbf{x}^{(1)} = (230.1,\, 37.8,\, 69.2)^\top$ and $y^{(1)} = 22.1$
        \item Stacking the $n$ observations gives the \textbf{feature matrix} $\mathbf{X} \in \mathbb{R}^{n \times p}$, whose row $i$ is $(\mathbf{x}^{(i)})^\top$, and the \textbf{label vector} $\mathbf{y} \in \mathbb{R}^{n}$
    \end{itemize}
    \[
        \mathbf{X} = \begin{bmatrix} 230.1 & 37.8 & 69.2 \\ 44.5 & 39.3 & 45.1 \\ \vdots & \vdots & \vdots \\ 232.1 & 8.6 & 8.7 \end{bmatrix} = \begin{bmatrix} (\mathbf{x}^{(1)})^\top \\ (\mathbf{x}^{(2)})^\top \\ \vdots \\ (\mathbf{x}^{(n)})^\top \end{bmatrix}, \qquad \mathbf{y} = \begin{bmatrix} 22.1 \\ 10.4 \\ \vdots \\ 13.4 \end{bmatrix}
    \]
    \begin{itemize}
        \item Superscript $(i)$ = which observation (row), subscript $j$ = which feature (column): $x_j^{(i)}$ is entry $(i, j)$ of $\mathbf{X}$; bold lowercase = column vector, bold uppercase = matrix, plain letter = single number
    \end{itemize}
\end{frame}

\begin{frame}{The regression problem}
    \begin{itemize}
        \item Assume the label depends on the features through $y = f(\mathbf{x}) + \varepsilon$: $f$ is a fixed but unknown function and $\varepsilon$ is a random error with mean zero, independent of $\mathbf{x}$
        \item \textbf{Supervised learning}: estimate $f$ from the labelled data $\{(\mathbf{x}^{(i)}, y^{(i)})\}_{i=1}^{n}$, which gives an estimate $\hat{f}$
        \item \textbf{Regression} when the label is quantitative (sales, mpg, balance); classification when it is qualitative (later lectures)
        \item \textbf{Prediction}: compute $\hat{y} = \hat{f}(\mathbf{x})$ for new observations, treating $\hat{f}$ as a black box
        \item \textbf{Inference}: understand how the label changes with each feature (which features matter, how large the effects are, whether the relationship is linear)
    \end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Advertising data: questions regression can answer}
    \begin{center}
        \includegraphics[width=0.95\linewidth,keepaspectratio]{figures/reg_advertising_scatter.pdf}
    \end{center}
    \begin{itemize}
        \item Is there a relationship between the advertising budgets and sales, and how strong is it?
        \item Which media are associated with sales, and by how much does each contribute?
        \item How accurately can we predict sales for given budgets?
        \item Is the relationship linear, and do the media reinforce each other (synergy)?
    \end{itemize}
\end{frame}

\section{Simple Linear Regression}

\begin{frame}{Simple linear regression model}
    \begin{itemize}
        \item One feature $x$ and a quantitative label $y$, related by a straight line plus error: $y = \beta_0 + \beta_1 x + \varepsilon$
        \item $\beta_0$ is the \textbf{intercept} (the value of $y$ when $x = 0$), $\beta_1$ is the \textbf{slope} (the change in $y$ for a one-unit increase in $x$); together they are the \textbf{coefficients}
        \item Estimate the coefficients from the training data (hat = estimated): $\hat\beta_0, \hat\beta_1$; predict with $\hat{y} = \hat\beta_0 + \hat\beta_1 x$
        \item \textbf{Fitted value} of observation $i$: $\hat{y}^{(i)} = \hat\beta_0 + \hat\beta_1 x^{(i)}$; \textbf{residual}: $e^{(i)} = y^{(i)} - \hat{y}^{(i)}$; residuals are computed from data, the errors $\varepsilon^{(i)}$ are never observed
        \item Example: $\text{sales} \approx \beta_0 + \beta_1 \times \text{TV}$
    \end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Least squares criterion}
    \begin{center}
        \includegraphics[width=0.95\linewidth,keepaspectratio]{figures/reg_slr_ls.pdf}
    \end{center}
    \begin{itemize}
        \item \textbf{Residual sum of squares}: $\mathrm{RSS} = \sum_{i=1}^{n} \big(e^{(i)}\big)^2 = \sum_{i=1}^{n} \big(y^{(i)} - \beta_0 - \beta_1 x^{(i)}\big)^2$
        \item \textbf{Least squares} chooses $\hat\beta_0, \hat\beta_1$ to minimize RSS: the line with the smallest total squared vertical distance to the points
        \item Squaring stops positive and negative residuals from cancelling and penalizes large ones more; the red dot marks the minimum (RSS $= 2102.5$ for TV)
    \end{itemize}
\end{frame}

\begin{frame}{Least squares estimates}
    Setting the partial derivatives of RSS with respect to $\beta_0$ and $\beta_1$ to zero gives closed-form estimates, where $\bar{x}$ and $\bar{y}$ are the sample means:
    \[
        \hat\beta_1 = \frac{\sum_{i=1}^{n} \big(x^{(i)} - \bar{x}\big)\big(y^{(i)} - \bar{y}\big)}{\sum_{i=1}^{n} \big(x^{(i)} - \bar{x}\big)^2}, \qquad \hat\beta_0 = \bar{y} - \hat\beta_1 \bar{x}
    \]
    \begin{itemize}
        \item $\hat\beta_1$ is the sample covariance of $x$ and $y$ divided by the sample variance of $x$
        \item The fitted line always passes through the point of means $(\bar{x}, \bar{y})$
        \item Advertising, sales on TV: $\hat\beta_0 = 7.03$ and $\hat\beta_1 = 0.0475$
    \end{itemize}
\end{frame}

\mode<article>{
    \paragraph{Derivation of the least squares estimates.}
    The partial derivatives of RSS are
    \begin{align*}
        \frac{\partial\, \mathrm{RSS}}{\partial \beta_0} &= -2 \sum_{i=1}^{n} \big(y^{(i)} - \beta_0 - \beta_1 x^{(i)}\big), &
        \frac{\partial\, \mathrm{RSS}}{\partial \beta_1} &= -2 \sum_{i=1}^{n} x^{(i)} \big(y^{(i)} - \beta_0 - \beta_1 x^{(i)}\big).
    \end{align*}
    Setting the first to zero gives $\bar{y} = \beta_0 + \beta_1 \bar{x}$, so $\hat\beta_0 = \bar{y} - \hat\beta_1 \bar{x}$. Substituting this into the second equation gives $\sum_i x^{(i)} (y^{(i)} - \bar{y}) = \hat\beta_1 \sum_i x^{(i)} (x^{(i)} - \bar{x})$, and since $\sum_i x^{(i)} (y^{(i)} - \bar{y}) = \sum_i (x^{(i)} - \bar{x})(y^{(i)} - \bar{y})$ and $\sum_i x^{(i)} (x^{(i)} - \bar{x}) = \sum_i (x^{(i)} - \bar{x})^2$, the formula for $\hat\beta_1$ follows.
}

\begin{frame}{Interpreting the coefficients}
    \begin{itemize}
        \item Fitted line for the Advertising data: $\widehat{\text{sales}} = 7.03 + 0.0475 \times \text{TV}$ (sales in thousands of units, TV budget in thousands of dollars)
        \item $\hat\beta_1 = 0.0475$: an additional \$1,000 of TV budget is associated with about $0.0475 \times 1000 = 47.5$ additional units sold
        \item $\hat\beta_0 = 7.03$: about 7,030 units are predicted when the TV budget is zero; the smallest TV budget in the data is \$700, so the intercept sits at the edge of the data
        \item The fit describes association in the observed markets; a causal reading needs more than a fitted line, because budgets were not assigned at random
    \end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{How accurate are the estimates?}
    \begin{center}
        \includegraphics[width=0.95\linewidth,keepaspectratio]{figures/reg_slr_sampling.pdf}
    \end{center}
    \begin{itemize}
        \item The population regression line (red) is the true, unknown relationship; the least squares line (blue) is computed from one sample, and every new sample gives a different line
        \item On average the estimates are correct (unbiased), but a single $\hat\beta_1$ differs from $\beta_1$; the \textbf{standard error} measures the typical size of this difference
    \end{itemize}
\end{frame}

\begin{frame}{Standard errors of the coefficients}
    \[
        \mathrm{SE}(\hat\beta_1)^2 = \frac{\sigma^2}{\sum_{i=1}^{n} \big(x^{(i)} - \bar{x}\big)^2}, \qquad \mathrm{SE}(\hat\beta_0)^2 = \sigma^2 \bigg[\frac{1}{n} + \frac{\bar{x}^2}{\sum_{i=1}^{n} \big(x^{(i)} - \bar{x}\big)^2}\bigg]
    \]
    \begin{itemize}
        \item Here $\sigma^2 = \mathrm{Var}(\varepsilon)$ is the variance of the errors; standard errors shrink with more data, less noise, and more spread-out $x^{(i)}$
        \item $\sigma^2$ is unknown and is estimated from the residuals by $\mathrm{RSE}^2 = \mathrm{RSS}/(n-2)$; replacing $\sigma$ by RSE gives the estimated standard errors reported by software
        \item These formulas assume the errors are uncorrelated with constant variance $\sigma^2$ (checked with residual plots later)
        \item Advertising, sales on TV: $\mathrm{SE}(\hat\beta_0) = 0.4578$ and $\mathrm{SE}(\hat\beta_1) = 0.0027$
    \end{itemize}
\end{frame}

\begin{frame}{Confidence intervals and hypothesis tests (recap)}
    \begin{itemize}
        \item \textbf{95\% confidence interval} for $\beta_1$: approximately $\hat\beta_1 \pm 2 \cdot \mathrm{SE}(\hat\beta_1)$ (exactly, the $0.975$ quantile of the $t_{n-2}$ distribution replaces 2)
        \item \textbf{Hypothesis test}: $H_0\colon \beta_1 = 0$ (no relationship) against $H_a\colon \beta_1 \neq 0$, using $t = \hat\beta_1 / \mathrm{SE}(\hat\beta_1)$, which follows a $t_{n-2}$ distribution if $H_0$ is true
        \item The \textbf{$p$-value} is the probability of a $|t|$ at least this large if $H_0$ were true; a small $p$-value (say below 0.05) is evidence against $H_0$
        \item Advertising, sales on TV: 95\% confidence interval for $\beta_1$ is $[0.0422,\, 0.0528]$
    \end{itemize}
    \begin{center}
        \small
        \begin{tabular}{lrrrr}
            \toprule
            & coefficient & std. error & $t$-statistic & $p$-value \\
            \midrule
            intercept & 7.0326 & 0.4578 & 15.36 & $<0.0001$ \\
            TV & 0.0475 & 0.0027 & 17.67 & $<0.0001$ \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}{Assessing the fit: RSE and R-squared}
    \begin{itemize}
        \item \textbf{Residual standard error}: $\mathrm{RSE} = \sqrt{\mathrm{RSS}/(n-2)}$, the typical size of a residual, in the units of $y$; Advertising: $\mathrm{RSE} = 3.26$, about 3,260 units (23\% of the mean sales)
        \item \textbf{$R^2$}: the proportion of the variation in $y$ explained by the regression
    \end{itemize}
    \[
        R^2 = 1 - \frac{\mathrm{RSS}}{\mathrm{TSS}}, \qquad \mathrm{TSS} = \sum_{i=1}^{n} \big(y^{(i)} - \bar{y}\big)^2
    \]
    \begin{itemize}
        \item $0 \le R^2 \le 1$; Advertising: $R^2 = 1 - 2102.5/5417.2 = 0.612$, so TV explains about 61\% of the variance in sales
        \item In simple regression $R^2 = r^2$, the squared correlation between $x$ and $y$ ($r = 0.782$)
        \item RSE is in the units of $y$, $R^2$ is unit-free; a good $R^2$ depends on the application
    \end{itemize}
\end{frame}


\section{Multiple Linear Regression}

\begin{frame}{Why not several simple regressions?}
    \begin{center}
        \small
        \begin{tabular}{lrrrr}
            \toprule
            sales on & $\hat\beta_1$ & std. error & $t$-statistic & $p$-value \\
            \midrule
            TV & 0.0475 & 0.0027 & 17.67 & $<0.0001$ \\
            radio & 0.2025 & 0.0204 & 9.92 & $<0.0001$ \\
            newspaper & 0.0547 & 0.0166 & 3.30 & 0.0011 \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{itemize}
        \item Three separate simple regressions give no single prediction from all three budgets, and each one ignores the other media
        \item A simple-regression slope also absorbs the effect of every correlated feature that was left out
        \item Newspaper and radio budgets are correlated ($r = 0.354$): markets that spend more on radio tend to spend more on newspaper, so newspaper may look useful only because it stands in for radio
        \item Remedy: put all features in one model
    \end{itemize}
\end{frame}


\begin{frame}{Multiple linear regression model}
    \begin{itemize}
        \item With $p$ features: $y = \beta_0 + \beta_1 x_1 + \dots + \beta_p x_p + \varepsilon$; $\beta_j$ is the average change in $y$ for a one-unit increase in $x_j$, \emph{holding all other features fixed}
        \item For all $n$ observations at once, $\mathbf{y} = \tilde{\mathbf{X}} \symbf{\beta} + \symbf{\varepsilon}$:
    \end{itemize}
    \[
        \underbrace{\begin{bmatrix} y^{(1)} \\ \vdots \\ y^{(n)} \end{bmatrix}}_{\mathbf{y}}
        = \underbrace{\begin{bmatrix} 1 & x_1^{(1)} & \cdots & x_p^{(1)} \\ \vdots & \vdots & & \vdots \\ 1 & x_1^{(n)} & \cdots & x_p^{(n)} \end{bmatrix}}_{\tilde{\mathbf{X}}}
        \underbrace{\begin{bmatrix} \beta_0 \\ \beta_1 \\ \vdots \\ \beta_p \end{bmatrix}}_{\symbf{\beta}}
        + \underbrace{\begin{bmatrix} \varepsilon^{(1)} \\ \vdots \\ \varepsilon^{(n)} \end{bmatrix}}_{\symbf{\varepsilon}}
    \]
    \begin{itemize}
        \item $\tilde{\mathbf{X}} = [\,\mathbf{1} \;\; \mathbf{X}\,] \in \mathbb{R}^{n \times (p+1)}$ is the feature matrix with a column of ones for the intercept (the \textbf{design matrix}); fitted values $\hat{\mathbf{y}} = \tilde{\mathbf{X}} \hat{\symbf{\beta}}$, residuals $\mathbf{e} = \mathbf{y} - \hat{\mathbf{y}}$
    \end{itemize}
\end{frame}


\begin{frame}{Least squares in matrix form}
    \begin{itemize}
        \item The RSS is a function of the coefficient vector: $\mathrm{RSS}(\symbf{\beta}) = \sum_{i=1}^{n} \big(y^{(i)} - \beta_0 - \beta_1 x_1^{(i)} - \dots - \beta_p x_p^{(i)}\big)^2 = \|\mathbf{y} - \tilde{\mathbf{X}} \symbf{\beta}\|^2$
        \item Setting its gradient to zero gives the \textbf{normal equations} $\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}}\, \hat{\symbf{\beta}} = \tilde{\mathbf{X}}^\top \mathbf{y}$, with solution
    \end{itemize}
    \[
        \hat{\symbf{\beta}} = \big(\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}}\big)^{-1} \tilde{\mathbf{X}}^\top \mathbf{y}
    \]
    \begin{itemize}
        \item The inverse exists when the $p+1$ columns of $\tilde{\mathbf{X}}$ are linearly independent: this needs $n > p$ and no feature that is an exact linear combination of the others
        \item Standard errors: $\widehat{\mathrm{Var}}(\hat{\symbf{\beta}}) = \hat\sigma^2 \big(\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}}\big)^{-1}$ with $\hat\sigma^2 = \mathrm{RSS}/(n-p-1)$; $\mathrm{SE}(\hat\beta_j)$ is the square root of the diagonal entry for $\beta_j$
    \end{itemize}
\end{frame}


\mode<article>{
    \paragraph{Derivation of the normal equations.}
    Expanding the squared norm gives
    \[
        \mathrm{RSS}(\symbf{\beta}) = \mathbf{y}^\top \mathbf{y} - 2\, \symbf{\beta}^\top \tilde{\mathbf{X}}^\top \mathbf{y} + \symbf{\beta}^\top \tilde{\mathbf{X}}^\top \tilde{\mathbf{X}} \symbf{\beta},
        \qquad
        \nabla_{\symbf{\beta}}\, \mathrm{RSS} = -2\, \tilde{\mathbf{X}}^\top \mathbf{y} + 2\, \tilde{\mathbf{X}}^\top \tilde{\mathbf{X}} \symbf{\beta}.
    \]
    Setting the gradient to zero gives $\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}} \hat{\symbf{\beta}} = \tilde{\mathbf{X}}^\top \mathbf{y}$. Equivalently, $\tilde{\mathbf{X}}^\top (\mathbf{y} - \tilde{\mathbf{X}} \hat{\symbf{\beta}}) = \mathbf{0}$: the residual vector $\mathbf{e}$ is orthogonal to every column of $\tilde{\mathbf{X}}$, in particular its entries sum to zero (the column of ones) and are uncorrelated with each feature. For $p = 1$ the solution reduces to the simple regression formulas.
}


\begin{frame}{Interpreting the coefficients}
    \begin{center}
        \small
        \begin{tabular}{lrrrr}
            \toprule
            & coefficient & std. error & $t$-statistic & $p$-value \\
            \midrule
            intercept & 2.9389 & 0.3119 & 9.42 & $<0.0001$ \\
            TV & 0.0458 & 0.0014 & 32.81 & $<0.0001$ \\
            radio & 0.1885 & 0.0086 & 21.89 & $<0.0001$ \\
            newspaper & $-0.0010$ & 0.0059 & $-0.18$ & 0.860 \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{itemize}
        \item Holding radio and newspaper fixed, an extra \$1,000 on TV is associated with about 45.8 more units sold; an extra \$1,000 on radio with about 188.5 more units
        \item Newspaper is not significant ($p = 0.86$) once TV and radio are in the model, although it was significant on its own ($p = 0.0011$)
        \item The correlation between newspaper and radio explains it: the simple regression credited newspaper with the effect of radio
    \end{itemize}
\end{frame}


\begin{frame}{Is at least one feature useful? The F-test}
    \begin{itemize}
        \item $H_0\colon \beta_1 = \beta_2 = \dots = \beta_p = 0$ against $H_a\colon$ at least one $\beta_j \neq 0$
    \end{itemize}
    \[
        F = \frac{(\mathrm{TSS} - \mathrm{RSS})/p}{\mathrm{RSS}/(n-p-1)}
    \]
    \begin{itemize}
        \item If $H_0$ is true, $F$ is close to 1; a value much larger than 1 is evidence against $H_0$, with the $p$-value taken from the $F_{p,\,n-p-1}$ distribution
        \item Advertising: $F = 570.3$ on $(3, 196)$ degrees of freedom, $p < 0.0001$: at least one medium is related to sales
        \item Why not just look at the individual $t$-tests? With many features, about 5\% of the $p$-values fall below 0.05 by chance even if no feature matters; the $F$-test accounts for the number of features
        \item Deciding \emph{which} features to keep is variable selection, deferred to a later lecture
    \end{itemize}
\end{frame}


\begin{frame}{Model fit in multiple regression}
    \begin{center}
        \small
        \begin{tabular}{lrr}
            \toprule
            model & $R^2$ & RSE \\
            \midrule
            TV & 0.612 & 3.26 \\
            radio & 0.332 & 4.27 \\
            TV + radio & 0.897 & 1.68 \\
            TV + radio + newspaper & 0.897 & 1.69 \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{itemize}
        \item $\mathrm{RSE} = \sqrt{\mathrm{RSS}/(n-p-1)}$ and $R^2 = 1 - \mathrm{RSS}/\mathrm{TSS}$; here $R^2$ is also the squared correlation between $\mathbf{y}$ and $\hat{\mathbf{y}}$
        \item Adding radio to TV raises $R^2$ from 0.612 to 0.897 and cuts RSE from 3.26 to 1.68; adding newspaper changes nothing useful
        \item $R^2$ never decreases when a feature is added, useful or not, while RSE can increase (the divisor $n-p-1$ shrinks); adjusted $R^2$ corrects this (performance measures, later)
    \end{itemize}
\end{frame}


\begin{frame}{Prediction and its uncertainty}
    \begin{itemize}
        \item For a new observation $\mathbf{x}_0 = (x_{0,1}, \dots, x_{0,p})^\top$, with $\tilde{\mathbf{x}}_0 = (1, x_{0,1}, \dots, x_{0,p})^\top$, the prediction is
    \end{itemize}
    \[
        \hat{y}_0 = \hat\beta_0 + \hat\beta_1 x_{0,1} + \dots + \hat\beta_p x_{0,p} = \tilde{\mathbf{x}}_0^\top \hat{\symbf{\beta}}
    \]
    \begin{itemize}
        \item Uncertainty from estimating the coefficients (reducible): $\hat{f}$ differs from the true $f$; the \textbf{confidence interval} covers the \emph{average} response $f(\mathbf{x}_0)$ over many observations with these features
        \item Model bias: a linear model is an approximation of reality; the intervals assume it is correct (checked with residual plots later)
        \item Irreducible error: a single response is $y_0 = f(\mathbf{x}_0) + \varepsilon$ even if $f$ were known; the \textbf{prediction interval} covers an \emph{individual} response and is always wider than the confidence interval
    \end{itemize}
\end{frame}

\section{Qualitative Predictors and Interactions}

\begin{frame}{Qualitative features with two levels}
    \begin{itemize}
        \item Credit data: average credit card balance (dollars) of $n = 400$ cardholders; do women and men differ?
        \item Encode gender with a \textbf{dummy variable}: $x^{(i)} = 1$ if person $i$ is female, $0$ if male
        \item Model $y = \beta_0 + \beta_1 x + \varepsilon$: the mean balance is $\beta_0 + \beta_1$ for women and $\beta_0$ for men, so $\beta_0$ is the average for men (the \textbf{baseline}) and $\beta_1$ the average difference between women and men
        \item Credit fit: $\hat\beta_0 = 509.80$ (SE 33.13) and $\hat\beta_1 = 19.73$ (SE 46.05, $p = 0.669$): women carry about \$20 more, but the difference is not statistically significant
        \item Coding male as 1 and female as 0, or as $\pm 1$, changes what the coefficients mean but not the fitted values
    \end{itemize}
\end{frame}

\begin{frame}{Qualitative features with more than two levels}
    \begin{center}
        \small
        \begin{tabular}{lcc}
            \toprule
            ethnicity & $x_1$ (Asian) & $x_2$ (Caucasian) \\
            \midrule
            African American & 0 & 0 \\
            Asian & 1 & 0 \\
            Caucasian & 0 & 1 \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{itemize}
        \item A feature with $k$ levels needs $k-1$ dummy variables; the level without one is the baseline; one 0/1/2 variable would impose an order
        \item $y = \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \varepsilon$: $\beta_0$ is the baseline average, $\beta_1$ and $\beta_2$ are differences from the baseline
        \item Credit fit: $\hat\beta_0 = 531.00$, $\hat\beta_1 = -18.69$ ($p = 0.774$), $\hat\beta_2 = -12.50$ ($p = 0.826$)
        \item The individual $p$-values depend on the baseline; test the whole feature with the $F$-test: $F = 0.04$, $p = 0.957$
    \end{itemize}
\end{frame}

\begin{frame}{Interactions: relaxing the additive assumption}
    \begin{itemize}
        \item The additive model assumes the effect of TV on sales is the same at every radio budget; spending on both together may work better than the sum of the parts (synergy)
        \item Add an \textbf{interaction term}, a new column of $\tilde{\mathbf{X}}$ with entries $x_1^{(i)} x_2^{(i)}$: $y = \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \beta_3 x_1 x_2 + \varepsilon = \beta_0 + \beta_1 x_1 + (\beta_2 + \beta_3 x_1)\, x_2 + \varepsilon$, so the effect of $x_2$ depends on $x_1$
        \item Advertising: $\widehat{\text{sales}} = 6.75 + 0.0191\,\text{TV} + 0.0289\,\text{radio} + 0.0011\,\text{TV} \times \text{radio}$; the interaction has $p < 0.0001$, $R^2$ rises from 0.897 to 0.968 and RSE falls from 1.68 to 0.94
        \item An extra \$1,000 on radio is associated with about 28.9 more units when TV $= 0$ and about 137.5 more units when TV $= \$100{,}000$
        \item \textbf{Hierarchical principle}: if an interaction is in the model, keep the main effects even when their $p$-values are large
    \end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Interaction with a qualitative feature}
    \begin{center}

\section{Polynomial Regression}

\begin{frame}{Polynomial regression}
    \begin{itemize}
        \item In the Auto data, mpg is a curved function of horsepower, so a straight line misses systematic structure
        \item \textbf{Polynomial regression} adds powers of the feature: $y = \beta_0 + \beta_1 x + \beta_2 x^2 + \dots + \beta_d x^d + \varepsilon$
        \item It is still a \emph{linear} regression, linear in the coefficients, with $x, x^2, \dots, x^d$ treated as $d$ features: least squares, standard errors and diagnostics carry over unchanged
    \end{itemize}
    \[
        \tilde{\mathbf{X}} = \begin{bmatrix} 1 & x^{(1)} & (x^{(1)})^2 & \cdots & (x^{(1)})^d \\ \vdots & \vdots & \vdots & & \vdots \\ 1 & x^{(n)} & (x^{(n)})^2 & \cdots & (x^{(n)})^d \end{bmatrix}, \qquad \hat{\symbf{\beta}} = \big(\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}}\big)^{-1} \tilde{\mathbf{X}}^\top \mathbf{y}
    \]
    \begin{itemize}
        \item Quadratic fit ($d = 2$): $\widehat{\text{mpg}} = 56.90 - 0.4662\,\text{hp} + 0.001231\,\text{hp}^2$, both coefficients with $p < 0.0001$
    \end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Polynomial fits on the Auto data}
    \begin{center}
        \includegraphics[width=0.95\linewidth]{figures/reg_auto_poly.pdf}
    \end{center}
    \begin{itemize}
        \item $R^2$ is 0.606 for the linear fit, 0.688 for the quadratic and 0.697 for degree 5: the quadratic term helps a lot, further terms barely
        \item The linear fit leaves a U-shaped pattern in the residuals, a systematic structure it failed to capture; for the quadratic fit the U-shape is gone
        \item Higher degrees always improve the fit to the training data, but degree 5 starts to wiggle at the ends: too much flexibility is a risk (bias-variance trade-off, later)
    \end{itemize}
\end{frame}
