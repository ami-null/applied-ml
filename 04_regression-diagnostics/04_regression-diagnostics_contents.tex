\section{Regression Model and Its Assumptions}


\begin{frame}{The multiple linear regression model}
    \begin{itemize}
        \item We observe $n$ pairs $(\mathbf{x}^{(i)}, y^{(i)})$, $i = 1, \dots, n$, with $p$ features in each $\mathbf{x}^{(i)}$
        \item The model, written for each observation:
    \end{itemize}
    \[
        y^{(i)} = \beta_0 + \beta_1 x_1^{(i)} + \cdots + \beta_p x_p^{(i)} + \varepsilon^{(i)}
    \]
    \begin{itemize}
        \item In matrix form, $\mathbf{y} = \tilde{\mathbf{X}}\symbf{\beta} + \symbf{\varepsilon}$, where $\tilde{\mathbf{X}}$ is the $n \times (p+1)$ design matrix whose $i$-th row is $(1, x_1^{(i)}, \dots, x_p^{(i)})$
        \item The coefficients are estimated by least squares; inference (standard errors, intervals, tests) needs the assumptions on the next two slides
    \end{itemize}
\end{frame}


\begin{frame}{Assumptions of the MLR model (1/2): the mean and the features}
    \begin{itemize}
        \item \textbf{Linearity}: the mean of $y$ is linear in the coefficients, $\mathbb{E}[y^{(i)} \mid \mathbf{x}^{(i)}] = \beta_0 + \sum_{j=1}^{p} \beta_j x_j^{(i)}$
        \item \textbf{Zero-mean errors}: $\mathbb{E}[\varepsilon^{(i)} \mid \tilde{\mathbf{X}}] = 0$, so the errors are unrelated to the features
        \item \textbf{No perfect collinearity}: $\tilde{\mathbf{X}}$ has full column rank $p+1$, so $(\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1}$ exists and the coefficients are uniquely determined
    \end{itemize}
\end{frame}


\begin{frame}{Assumptions of the MLR model (2/2): the errors}
    \begin{itemize}
        \item \textbf{Uncorrelated errors}: $\mathrm{Cov}(\varepsilon^{(i)}, \varepsilon^{(k)}) = 0$ for $i \neq k$
        \item \textbf{Constant variance (homoscedasticity)}: $\mathrm{Var}(\varepsilon^{(i)}) = \sigma^2$ for every $i$
        \item \textbf{Normal errors}: $\varepsilon^{(i)} \sim \mathcal{N}(0, \sigma^2)$; needed for exact small-sample intervals and tests, not for the estimates themselves
    \end{itemize}
\end{frame}


\begin{frame}{These assumptions may not hold}
    \begin{itemize}
        \item Real data often violate one or more of them: the true relation is curved, errors are correlated (over time or within units), the spread of the errors changes with the level of $y$, a few observations are unusual, or features are nearly collinear
        \item Consequences: wrong predictions (non-linearity); wrong standard errors, intervals and $p$-values (correlated errors, non-constant variance, non-normal errors); unstable coefficients (collinearity); a few points distorting the whole fit (outliers, high leverage)
        \item So after fitting, the assumptions have to be checked, not assumed
    \end{itemize}
\end{frame}


\section{Model Diagnostics}


\begin{frame}{What is model diagnostics?}
    \begin{itemize}
        \item \textbf{Model diagnostics}: checks made after fitting to see whether the model assumptions hold, and whether the fit depends on a few observations or on redundant features
        \item The checks use quantities computed from the fit: residuals, studentized residuals, leverages and variance inflation factors
        \item Three questions: Is the linear model adequate? Which observations are unusual? Are some features redundant?
        \item Diagnostics examine whether the fit can be trusted; performance measures (later in this lecture) examine how well it fits
    \end{itemize}
\end{frame}


\begin{frame}{Coefficient estimates and fitted values}
    \begin{itemize}
        \item Least squares chooses $\symbf{\beta}$ to minimize $\mathrm{RSS} = \sum_{i=1}^{n} \big(y^{(i)} - \beta_0 - \sum_{j=1}^{p} \beta_j x_j^{(i)}\big)^2$, which gives
    \end{itemize}
    \[
        \hat{\symbf{\beta}} = (\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1} \tilde{\mathbf{X}}^\top \mathbf{y}
    \]
    \begin{itemize}
        \item The fitted values are obtained by plugging $\hat{\symbf{\beta}}$ into the model:
    \end{itemize}
    \[
        \hat{\mathbf{y}} = \tilde{\mathbf{X}}\hat{\symbf{\beta}}, \qquad \hat{y}^{(i)} = \hat\beta_0 + \sum_{j=1}^{p} \hat\beta_j x_j^{(i)}
    \]
\end{frame}


\begin{frame}{The hat matrix}
    \begin{itemize}
        \item Substituting $\hat{\symbf{\beta}}$ into $\hat{\mathbf{y}} = \tilde{\mathbf{X}}\hat{\symbf{\beta}}$ expresses the fitted values as a linear map of the labels:
    \end{itemize}
    \[
        \hat{\mathbf{y}} = \tilde{\mathbf{X}} (\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1} \tilde{\mathbf{X}}^\top \mathbf{y} = \mathbf{H}\mathbf{y}, \qquad \mathbf{H} = \tilde{\mathbf{X}} (\tilde{\mathbf{X}}^\top \tilde{\mathbf{X}})^{-1} \tilde{\mathbf{X}}^\top
    \]
    \begin{itemize}
        \item $\mathbf{H}$ is the \textbf{hat matrix}: $n \times n$, symmetric and idempotent ($\mathbf{H}^2 = \mathbf{H}$), with $\mathrm{tr}(\mathbf{H}) = p+1$
        \item $\mathbf{H}$ depends only on the features in $\tilde{\mathbf{X}}$, never on $\mathbf{y}$
    \end{itemize}
\end{frame}


\begin{frame}{The residual vector}
    \begin{itemize}
        \item The residual for observation $i$ is the gap between the observed and fitted value, $e^{(i)} = y^{(i)} - \hat{y}^{(i)}$; in vector form $\mathbf{e} = \mathbf{y} - \hat{\mathbf{y}}$
        \item Using $\hat{\mathbf{y}} = \mathbf{H}\mathbf{y}$, the residual vector is a linear function of $\mathbf{y}$:
    \end{itemize}
    \[
        \mathbf{e} = \mathbf{y} - \mathbf{H}\mathbf{y} = (\mathbf{I} - \mathbf{H})\,\mathbf{y}
    \]
    \begin{itemize}
        \item The residuals estimate the unobservable errors: $\varepsilon^{(i)}$ is never known, $e^{(i)}$ is computed from the data
    \end{itemize}
\end{frame}


\begin{frame}{The residual vector is a random vector}
    \begin{itemize}
        \item $\mathbf{e}$ is a function of the random errors $\symbf{\varepsilon}$, so each $e^{(i)}$ is a random variable and $\mathbf{e}$ is a random vector
        \item Its mean is $\mathbb{E}[\mathbf{e}] = \mathbf{0}$; its spread is described by the \textbf{variance--covariance matrix}
    \end{itemize}
    \[
        \mathrm{Cov}(\mathbf{e}) = \mathbb{E}\big[(\mathbf{e} - \mathbb{E}[\mathbf{e}])(\mathbf{e} - \mathbb{E}[\mathbf{e}])^\top\big] = \sigma^2 (\mathbf{I} - \mathbf{H})
    \]
    \begin{itemize}
        \item $\mathrm{Cov}(\mathbf{e})$ is $n \times n$ and symmetric
        \item Diagonal entries are the individual variances: $\mathrm{Var}(e^{(i)}) = \sigma^2 (1 - h^{(i)})$
        \item Off-diagonal entries are the pairwise covariances: $\mathrm{Cov}(e^{(i)}, e^{(k)}) = -\sigma^2 H_{ik}$ for $i \neq k$
        \item The errors are uncorrelated, but the residuals generally are not, because $H_{ik} \neq 0$
    \end{itemize}
\end{frame}


\begin{frame}{The variance--covariance matrix of the residuals}
    \begin{itemize}
        \item $\mathrm{Cov}(\mathbf{e}) = \sigma^2(\mathbf{I} - \mathbf{H})$ is an $n \times n$ symmetric matrix
    \end{itemize}
    \[
        \mathrm{Cov}(\mathbf{e}) = \sigma^2 \begin{pmatrix}
            \textcolor{blue}{1 - h^{(1)}} & -H_{12} & -H_{13} & \cdots & -H_{1n} \\
            -H_{21} & \textcolor{blue}{1 - h^{(2)}} & -H_{23} & \cdots & -H_{2n} \\
            -H_{31} & -H_{32} & \textcolor{blue}{1 - h^{(3)}} & \cdots & -H_{3n} \\
            \vdots & \vdots & \vdots & \ddots & \vdots \\
            -H_{n1} & -H_{n2} & -H_{n3} & \cdots & \textcolor{blue}{1 - h^{(n)}}
        \end{pmatrix}
    \]
    \begin{itemize}
        \item Blue diagonal: the individual variances $\mathrm{Var}(e^{(i)}) = \sigma^2(1 - h^{(i)})$
        \item Black off-diagonal: the pairwise covariances $\mathrm{Cov}(e^{(i)}, e^{(k)}) = -\sigma^2 H_{ik}$ for $i \neq k$
        % \item Every row sums to zero, because the intercept column makes $(\mathbf{I} - \mathbf{H})\tilde{\mathbf{X}} = \mathbf{0}$
    \end{itemize}
\end{frame}


\begin{frame}{Where the hat matrix is used: leverage}
    \begin{itemize}
        \item The diagonal of the covariance matrix (previous slides) gives the variance of each residual:
    \end{itemize}
    \[
        \mathrm{Var}\big(e^{(i)}\big) = \sigma^2 \big(1 - h^{(i)}\big), \qquad h^{(i)} = H_{ii}
    \]
    \begin{itemize}
        \item $h^{(i)}$ is the \textbf{leverage} of observation $i$; it depends only on the features, and the leverages add up to $\sum_{i=1}^{n} h^{(i)} = p+1$
        \item Even with constant error variance, residuals have unequal variances: high-leverage observations have smaller residual variance, because the fit is pulled towards them
    \end{itemize}
\end{frame}


\begin{frame}{What is residual analysis?}
    \begin{itemize}
        \item \textbf{Residual analysis}: examining the residuals $\mathbf{e}$ (or transformed versions of them) to judge whether the assumptions of the model hold
        \item If the assumptions hold, the residuals should look like random noise: centred at zero, with constant spread, no pattern against the fitted values, the features, or the observation order
        \item Main plots: residuals against fitted values, residuals against each feature, residuals against observation order, and the normal QQ plot of the residuals
        \item Observation-level quantities (leverage and studentized residuals) refine the check; they need the hat matrix, shown next
    \end{itemize}
\end{frame}


\begin{frame}{Where the hat matrix is used: leverage}
    \begin{itemize}
        \item Since $\mathbf{e} = (\mathbf{I} - \mathbf{H})\symbf{\varepsilon}$, we have $\mathbb{E}[\mathbf{e}] = \mathbf{0}$ and $\mathrm{Cov}(\mathbf{e}) = \sigma^2 (\mathbf{I} - \mathbf{H})$; reading off the diagonal,
    \end{itemize}
    \[
        \mathrm{Var}\big(e^{(i)}\big) = \sigma^2 \big(1 - h^{(i)}\big), \qquad h^{(i)} = H_{ii}
    \]
    \begin{itemize}
        \item $h^{(i)}$ is the \textbf{leverage} of observation $i$; it depends only on the features, and the leverages add up to $\sum_{i=1}^{n} h^{(i)} = p+1$
        \item Even with constant error variance, residuals have unequal variances: high-leverage observations have smaller residual variance, because the fit is pulled towards them
    \end{itemize}
\end{frame}


\begin{frame}{Where the hat matrix is used: studentized residuals}
    \begin{itemize}
        \item Raw residuals have unequal variances and carry the units of $y$; dividing by the estimated standard deviation fixes both:
    \end{itemize}
    \[
        r^{(i)} = \frac{e^{(i)}}{\hat\sigma \sqrt{1 - h^{(i)}}}, \qquad \hat\sigma = \mathrm{RSE}
    \]
    \begin{itemize}
        \item The leverage $h^{(i)} = H_{ii}$ comes from the hat matrix, so the studentized residual uses both the residual and the hat matrix
        \item Each $r^{(i)}$ has variance close to 1 and no units, so residuals can be compared across observations and across models
        \item With normal errors about 99.7\% of the $r^{(i)}$ fall within $\pm 3$: an observation with $|r^{(i)}| > 3$ is unusual, although among 400 observations one such value is expected by chance
        \item The studentized residual is also called the standardized residual; in R it is computed by \texttt{rstandard()}
    \end{itemize}
\end{frame}


\mode<article>{
    \paragraph{Externally studentized residuals.}
    A variant replaces $\hat\sigma$ by $\hat\sigma_{(i)}$, the estimate computed with observation $i$ left out. It removes the influence of a suspicious point on its own residual and follows a $t_{n-p-2}$ distribution exactly under normal errors. In R this is \texttt{rstudent()}; for moderate $n$ the two versions rarely lead to different conclusions.
}


\begin{frame}{Problem 1: non-linearity, what to check}
    \begin{itemize}
        \item \textbf{Plot}: residuals $e^{(i)}$ against fitted values $\hat{y}^{(i)}$, with a horizontal line at zero
        \item \textbf{If} the points follow a curve (U-shape or inverted U), \textbf{then} the linearity assumption is violated
        \item \textbf{If} the points scatter randomly around zero with no pattern, \textbf{then} linearity is plausible
        \item \textbf{Consequence}: predictions are biased in the ranges where the curve lies, and inference is unreliable
        \item \textbf{Remedies}: transform features ($\log x$, $\sqrt{x}$), add interaction terms, or fit a more flexible model
    \end{itemize}
\end{frame}


\begin{frame}[allowframebreaks]{Problem 1: non-linearity, example}
    \begin{center}
        \includegraphics[width=0.95\linewidth]{figures/reg_diag_nonlinear.pdf}
    \end{center}
    \begin{itemize}
        \item Dataset: Advertising, $n = 200$ markets (sales in thousands of units; TV and radio budgets in thousands of dollars)
        \item Left panel: additive model, $\text{sales} = \beta_0 + \beta_1\,\text{TV} + \beta_2\,\text{radio} + \varepsilon$; right panel: the same model with a TV $\times$ radio term
        \item Additive model: residuals are positive at the smallest and largest fitted values and negative in between, a U-shape that signals missing synergy between TV and radio
        \item With the interaction the curvature weakens and RSE falls from 1.68 to 0.94; a few large negative residuals remain
    \end{itemize}
\end{frame}


\begin{frame}{Problem 2: correlated errors, what to check}
    \begin{itemize}
        \item \textbf{Plot}: residuals $e^{(t)}$ against observation order $t$, with a horizontal line at zero
        \item \textbf{If} neighbouring residuals are alike, with long runs of the same sign, \textbf{then} the uncorrelated-errors assumption is violated (successive errors are positively correlated)
        \item \textbf{If} the signs change at random from one observation to the next, \textbf{then} there is no evidence of correlation
        \item \textbf{Consequence}: standard errors are too small, so intervals are too narrow and $p$-values too small
        \item \textbf{Remedy}: time-series or mixed models, which are outside the scope of this lecture
    \end{itemize}
\end{frame}


\begin{frame}[allowframebreaks]{Problem 2: correlated errors, example}
    \begin{center}
        \includegraphics[width=0.95\linewidth]{figures/reg_diag_autocorr.pdf}
    \end{center}
    \begin{itemize}
        \item Dataset (simulated): $n = 100$ observations from $y^{(t)} = 1 + 0.05\,t + \varepsilon^{(t)}$, fitted by least squares with a linear trend in $t$
        \item Errors follow an AR(1) process, $\varepsilon^{(t)} = \rho\,\varepsilon^{(t-1)} + u^{(t)}$, with $\rho = 0$, $0.5$, $0.9$; here $\rho$ is the correlation of successive errors
        \item $\rho = 0$ gives random scatter around zero; $\rho = 0.9$ gives long runs above and below zero
    \end{itemize}
\end{frame}


\begin{frame}{Problem 3: non-constant variance, what to check}
    \begin{itemize}
        \item \textbf{Plot}: residuals $e^{(i)}$ against fitted values $\hat{y}^{(i)}$
        \item \textbf{If} the vertical spread grows or shrinks with the fitted value (a funnel), \textbf{then} the constant-variance assumption is violated
        \item \textbf{If} the vertical spread is about the same across the whole range, \textbf{then} constant variance is plausible
        \item \textbf{Consequence}: standard errors, intervals and tests become unreliable
        \item \textbf{Remedy 1}: transform $y$ ($\log y$ or $\sqrt{y}$)
        \item \textbf{Remedy 2}: weighted least squares with $w^{(i)} \propto 1/\mathrm{Var}(\varepsilon^{(i)})$, giving $\hat{\symbf{\beta}} = (\tilde{\mathbf{X}}^\top \mathbf{W} \tilde{\mathbf{X}})^{-1} \tilde{\mathbf{X}}^\top \mathbf{W} \mathbf{y}$ with $\mathbf{W} = \mathrm{diag}(w^{(i)})$
    \end{itemize}
\end{frame}


\begin{frame}[allowframebreaks]{Problem 3: non-constant variance, example}
    \begin{center}
        \includegraphics[width=0.95\linewidth]{figures/reg_diag_variance.pdf}
    \end{center}
    \begin{itemize}
        \item Dataset and model: Auto, $n = 392$ cars; $\text{mpg} = \beta_0 + \beta_1\,\text{horsepower} + \beta_2\,\text{weight} + \beta_3\,\text{year} + \varepsilon$ ($p = 3$)
        \item Left panel: $\log(\text{mpg})$ as response; right panel: mpg as response; the y-scales are free
        \item The mpg residuals fan out as the fitted value grows; their standard deviation grows from 2.35 to 2.48, 3.47 and 3.88 across the four quartiles of the fitted values
        \item For $\log(\text{mpg})$ the spread is nearly constant (0.116, 0.117, 0.130, 0.107), so the coefficients describe multiplicative effects
    \end{itemize}
\end{frame}


\begin{frame}{Problem 4: outliers, what to check}
    \begin{itemize}
        \item \textbf{Plot}: studentized residuals $r^{(i)}$ against fitted values $\hat{y}^{(i)}$, with horizontal lines at $\pm 3$
        \item \textbf{If} a point lies beyond $\pm 3$, \textbf{then} it is a suspected outlier: its error is far outside the pattern shown by the rest of the data
        \item \textbf{Consequence}: an outlier inflates RSE and lowers $R^2$, and it can pull the fit towards itself
        \item \textbf{Action}: investigate the point (data-entry error or a genuine case) before deleting anything; refit and compare
    \end{itemize}
\end{frame}


\begin{frame}[allowframebreaks]{Problem 4: outliers, example}
    \begin{center}
        \includegraphics[width=0.95\linewidth]{figures/reg_diag_outlier.pdf}
    \end{center}
    \begin{itemize}
        \item Dataset (simulated): 50 points with $x \sim U(-2, 2)$ and $y = 1 + 2x + \varepsilon$, $\varepsilon \sim N(0, 0.7^2)$, plus one added point at $x = 0.3$, $y = 7.6$, six units above the line
        \item Left: fit with (red) and without (blue) the added point; right: studentized residuals against fitted values, with the outlier in red
        \item Real data, Auto (model of Problem 3): 6 of 392 cars have $|r^{(i)}| > 3$; the largest, $r^{(i)} = 4.20$, is the most economical car in the data (46.6 mpg), far above its predicted value of about 32 mpg
    \end{itemize}
\end{frame}


\begin{frame}{Problem 5: high-leverage observations, what to check}
    \begin{itemize}
        \item \textbf{Plot}: studentized residuals $r^{(i)}$ against leverage $h^{(i)}$, with a vertical line at the average $(p+1)/n$ and horizontal lines at $\pm 3$
        \item \textbf{If} points lie far to the right of the average-leverage line, \textbf{then} they have high leverage: unusual feature values that can pull the fit towards themselves
        \item \textbf{If} a high-leverage point also has a large residual, \textbf{then} it is \emph{influential}: removing it changes the fit a lot
        \item High leverage alone is not a violation of an assumption; the plot checks whether one point controls the fit
        \item \textbf{Remedy}: check the point, refit without it, and report both fits
    \end{itemize}
\end{frame}


\begin{frame}[allowframebreaks]{Problem 5: high-leverage observations, example}
    \begin{center}
        \includegraphics[width=0.95\linewidth]{figures/reg_diag_leverage.pdf}
    \end{center}
    \begin{itemize}
        \item Dataset (simulated): the 50 points of Problem 4 without the outlier, plus one added point at $x = 5$, $y = 2$; the model $y = \beta_0 + \beta_1 x + \varepsilon$ is fitted to all 51 points
        \item Left: fit with (red) and without (blue) the added point; right: studentized residuals against leverage
        \item The added point has the most extreme $x$, so its leverage is high; its studentized residual is also large and negative, so it is influential and the fitted line tilts visibly towards it
        \item Real data, Auto (model of Problem 3): the largest leverage is $h^{(i)} = 0.095$ (a 225-horsepower car), 9.3 times the average $4/392$
    \end{itemize}
\end{frame}


\begin{frame}{Problem 6: collinearity, what to check}
    \begin{itemize}
        \item \textbf{Plot}: a scatterplot of each pair of features $x_j$ against $x_k$, with the correlation $r_{jk}$ in the title
        \item \textbf{If} the points lie almost on a straight line ($|r_{jk}|$ close to 1), \textbf{then} the two features carry nearly the same information, and the no-perfect-collinearity assumption is nearly violated
        \item Confirm with the VIF: values above 5 or 10 signal collinearity
        \item \textbf{Consequence}: unstable coefficients, inflated standard errors, and important features that look insignificant
        \item \textbf{Remedy}: drop one of the redundant features, or combine them
    \end{itemize}
\end{frame}


\begin{frame}[allowframebreaks]{Problem 6: collinearity, example}
    \begin{center}
        \includegraphics[width=0.95\linewidth]{figures/reg_credit_collinearity.pdf}
    \end{center}
    \begin{itemize}
        \item Dataset: Credit, $n = 400$ customers; response Balance, with features limit, rating and age
        \item Left: limit against age, $r = 0.10$, a diffuse cloud; right: limit against rating, $r = 0.997$, points lying almost on a line
        \item For Balance on age + limit, $\mathrm{SE}(\hat\beta_{\text{limit}}) = 0.0050$ ($p < 0.0001$); for rating + limit, $0.0638$, 13 times larger, with $p = 0.70$
    \end{itemize}
\end{frame}


\begin{frame}{Detecting collinearity: variance inflation factor}
    \begin{itemize}
        \item Pairwise correlations can miss collinearity among three or more features (multicollinearity); the \textbf{variance inflation factor} does not
    \end{itemize}
    \[
        \mathrm{VIF}(\hat\beta_j) = \frac{1}{1 - R^2_{x_j \mid x_{-j}}}
    \]
    \begin{itemize}
        \item $R^2_{x_j \mid x_{-j}}$ is the $R^2$ from regressing feature $x_j$ on all the other features; VIF is at least 1, and values above 5 or 10 signal a problem
        \item Credit, age + rating + limit: VIF $= 1.01$ (age), $160.7$ (rating), $160.6$ (limit); with only age and limit: $1.01$ and $1.01$
        \item Remedies: drop one of the redundant features, or combine them (for example, average the standardized features)
        \item Predictions barely change ($R^2 = 0.750$ and $0.746$ for the two models), but individual coefficients and their inference do
    \end{itemize}
\end{frame}


\begin{frame}{Problem 7: non-normal errors, what to check}
    \begin{itemize}
        \item \textbf{Plot}: normal QQ plot of the studentized residuals $r^{(i)}$: sorted residuals against the quantiles of $\mathcal{N}(0,1)$, with a reference line
        \item \textbf{If} the points follow the line, \textbf{then} normal errors are plausible
        \item \textbf{If} both ends bend away from the line, \textbf{then} the tails are heavier than normal: there are more extreme residuals than the normal model allows
        \item \textbf{If} only one end bends away, \textbf{then} the errors are skewed
        \item \textbf{Consequence}: exact small-sample intervals and tests are unreliable; with large $n$ the central limit theorem keeps estimates and inference roughly valid
        \item \textbf{Remedies}: transform $y$, use robust regression, or bootstrap the inference
    \end{itemize}
\end{frame}

\begin{frame}{Diagnostic checklist}
    \begin{center}
        \footnotesize
        \renewcommand{\arraystretch}{1.1}
        \setlength{\tabcolsep}{4pt}
        \begin{tabular}{llp{5cm}}
            \toprule
            Problem & Symptom & Remedy \\
            \midrule
            Non-linearity & Curve in residuals vs fitted & Transform features, interactions \\
            Correlated errors & Tracking of residuals over time & Time-series or mixed models \\
            Non-constant variance & Funnel in residuals vs fitted & Transform $y$, weighted least squares \\
            Outliers & $|r^{(i)}| > 3$ & Investigate, fix or remove errors \\
            High leverage & $h^{(i)}$ far above $(p+1)/n$, large $|r^{(i)}|$ & Check the point, refit without it \\
            Collinearity & VIF above 5 or 10 & Drop or combine features \\
            Non-normal errors & Both ends or one end off the line in QQ plot & Transform $y$, robust or bootstrap inference \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{itemize}
        \item Check in this order: residual plots first, then normality, then unusual observations, then collinearity; every fix changes the model, so refit and repeat the diagnostics
    \end{itemize}
\end{frame}


\section{Regression Performance Measures}


\begin{frame}{Measures of the quality of fit}
    \begin{center}
        \footnotesize
        \renewcommand{\arraystretch}{1.05}
        \begin{tabular}{lll}
            \toprule
            measure & definition & comment \\
            \midrule
            MSE & $\mathrm{RSS}/(n-p-1)$ & Squared units of $y$ \\
            RSE & $\sqrt{MSE}$ & Units of $y$, estimates $\sigma$ \\
            MAE & $\frac{1}{n} \sum_{i=1}^{n} \big|y^{(i)} - \hat{y}^{(i)}\big|$ & Units of $y$, less sensitive to outliers \\
            $R^2$ & $1 - \mathrm{RSS}/\mathrm{TSS}$ & Unit-free, never falls with more features \\
            Adjusted $R^2$ & $1 - \dfrac{\mathrm{RSS}/(n-p-1)}{\mathrm{TSS}/(n-1)}$ & Unit-free, penalizes each added feature \\
            \bottomrule
        \end{tabular}
    \end{center}
    where \(\mathrm{RSS} = \sum_{i=1}^{n} \big(y^{(i)} - \hat{y}^{(i)}\big)^2\).
    \begin{itemize}
        \item MSE and RSE punish large errors heavily (squares); MAE weighs all errors linearly, so a few outliers move it less
    \end{itemize}
\end{frame}


\begin{frame}{Interpreting $R^2$}
    \begin{itemize}
        \item $R^2 = 1 - \mathrm{RSS}/\mathrm{TSS} = \mathrm{ESS}/\mathrm{TSS}$ is also the squared correlation between $\mathbf{y}$ and $\hat{\mathbf{y}}$, and equals $r^2$ in simple regression
        \item It lies in $[0, 1]$ for a model with an intercept, and it never decreases when a feature is added: the larger model can always reproduce the smaller one by setting the new coefficient to zero
        \item It depends on the spread of the data: the same model fitted where $x$ (and hence $y$) varies over a wider range has a higher $R^2$, and $R^2$ values of models with different responses (for example mpg and $\log$ mpg) are not comparable
        \item High $R^2$ does not mean the model is right (check the residuals); low $R^2$ does not mean it is useless, since real effects can be small relative to noise, for example in survey or behavioural data
    \end{itemize}
\end{frame}


\begin{frame}{Adjusted R-squared}
    \[
        R^2_{\text{adj}} = 1 - \frac{\mathrm{RSS}/(n-p-1)}{\mathrm{TSS}/(n-1)} = 1 - (1 - R^2)\, \frac{n-1}{n-p-1}
    \]
    \begin{itemize}
        \item It compares two unbiased variance estimates: the residual variance $\hat\sigma^2$ and the variance of $y$; the factor $(n-1)/(n-p-1)$ grows with $p$ and penalizes each added feature
        \item Adding a feature raises $R^2_{\text{adj}}$ exactly when the feature's $|t|$-statistic exceeds 1 (equivalently, its partial $F$ exceeds 1); a weak bar, but it stops $R^2_{\text{adj}}$ from rising for pure noise
        \item Advertising: adding newspaper has $|t| = 0.18 < 1$, so $R^2_{\text{adj}}$ falls from 0.8962 to 0.8956 while $R^2$ stays 0.8972
    \end{itemize}
\end{frame}

\mode<article>{
    \paragraph{Why the threshold is $|t| > 1$.}
    Let $\mathrm{RSS}_{\text{old}}$ be the residual sum of squares before adding one feature and $\mathrm{RSS}_{\text{new}}$ afterwards, with $p$ features in the old model. Then $R^2_{\text{adj}}$ increases if and only if $\mathrm{RSS}_{\text{new}}/(n-p-2) < \mathrm{RSS}_{\text{old}}/(n-p-1)$. Rearranging, this is $1 < (\mathrm{RSS}_{\text{old}} - \mathrm{RSS}_{\text{new}})(n-p-2)/\mathrm{RSS}_{\text{new}}$, and the right-hand side is the partial $F$-statistic of the added feature, which equals its squared $t$-statistic.
}

\begin{frame}{Summary}
    \begin{itemize}
        \item \textbf{Assumptions}: linearity, zero-mean errors, full column rank, uncorrelated errors, constant variance and (for exact inference) normal errors; these may not hold, so they must be checked
        \item \textbf{Residuals}: $\mathbf{e} = (\mathbf{I} - \mathbf{H})\mathbf{y}$, with $\mathrm{Var}(e^{(i)}) = \sigma^2 (1 - h^{(i)})$; studentized residuals $r^{(i)} = e^{(i)}/(\hat\sigma\sqrt{1-h^{(i)}})$
        \item \textbf{Diagnostics}: residual plots reveal non-linearity, correlated errors and non-constant variance; studentized residuals flag outliers, leverage flags unusual feature values, VIF flags collinearity
        \item \textbf{Performance}: MSE, RMSE, MAE, RSE, $R^2$ and adjusted $R^2$ measure the fit to the training data; RSE and adjusted $R^2$ account for the degrees of freedom; training error keeps falling as features are added, test error does not
    \end{itemize}
\end{frame}
